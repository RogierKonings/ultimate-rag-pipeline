# US-10.2.3: Ingestion Rate Limiting

> **Initiative:** 2 - End-to-End Resilience & Adaptive Degradation
> **Priority:** P1
> **Effort:** Medium (2-3 days)
> **Status:** Implemented

## Summary

Implement per-tenant rate limiting on ingestion jobs to prevent noisy tenants from starving others and ensure fair resource allocation.

## Problem Statement

**Current state**: The ingestion service in [tasks/ingest.py](services/ingestion/tasks/ingest.py) processes jobs without tenant-based limits:
- All tenants share the same Celery queues
- A tenant with many jobs can monopolize workers
- No mechanism to prioritize or throttle
- Difficult to guarantee SLAs per tenant

**Impact**:
- Large batch ingestions block other tenants
- No fairness guarantees
- Resource planning is difficult
- Cannot offer tiered service levels

## User Story

**As a** platform operator
**I want** per-tenant rate limits on ingestion jobs
**So that** noisy tenants don't starve others

### Rate Limiter Implementation

```python
# services/ingestion/rate_limiting/limiter.py

from dataclasses import dataclass
import redis.asyncio as redis
import structlog

logger = structlog.get_logger(__name__)

@dataclass
class TenantLimits:
    """Rate limit configuration for a tenant."""
    max_concurrent_jobs: int = 10
    priority: str = "normal"  # high, normal, low
    hard_limit: bool = False  # If True, reject instead of queue

class IngestionRateLimiter:
    """
    Per-tenant rate limiting for ingestion jobs.

    Uses Redis for distributed coordination across workers.
    """

    def __init__(
        self,
        redis_client: redis.Redis,
        default_max_concurrent: int = 10,
    ):
        self.redis = redis_client
        self.default_max = default_max_concurrent
        self._key_prefix = "ingestion:rate_limit"

    def _active_jobs_key(self, tenant_id: str) -> str:
        return f"{self._key_prefix}:active:{tenant_id}"

    def _limits_key(self, tenant_id: str) -> str:
        return f"{self._key_prefix}:limits:{tenant_id}"

    def _queue_key(self, tenant_id: str) -> str:
        return f"{self._key_prefix}:queued:{tenant_id}"

    async def get_tenant_limits(self, tenant_id: str) -> TenantLimits:
        """Get rate limit configuration for tenant."""
        data = await self.redis.hgetall(self._limits_key(tenant_id))
        if not data:
            return TenantLimits(max_concurrent_jobs=self.default_max)

        return TenantLimits(
            max_concurrent_jobs=int(data.get(b"max_concurrent", self.default_max)),
            priority=data.get(b"priority", b"normal").decode(),
            hard_limit=data.get(b"hard_limit", b"false") == b"true",
        )

    async def set_tenant_limits(
        self,
        tenant_id: str,
        limits: TenantLimits,
    ) -> None:
        """Set rate limit configuration for tenant."""
        await self.redis.hset(
            self._limits_key(tenant_id),
            mapping={
                "max_concurrent": limits.max_concurrent_jobs,
                "priority": limits.priority,
                "hard_limit": str(limits.hard_limit).lower(),
            },
        )
        logger.info(
            "tenant_limits_updated",
            tenant_id=tenant_id,
            limits=limits,
        )

    async def try_acquire_slot(
        self,
        tenant_id: str,
        job_id: str,
    ) -> tuple[bool, str | None]:
        """
        Try to acquire a job slot for tenant.

        Returns:
            (acquired, reason) - True if slot acquired, else False with reason
        """
        limits = await self.get_tenant_limits(tenant_id)
        key = self._active_jobs_key(tenant_id)

        # Atomic check and increment
        async with self.redis.pipeline(transaction=True) as pipe:
            try:
                # Watch the key for changes
                await pipe.watch(key)
                current = await self.redis.scard(key)

                if current >= limits.max_concurrent_jobs:
                    await pipe.unwatch()
                    logger.info(
                        "rate_limit_reached",
                        tenant_id=tenant_id,
                        current=current,
                        limit=limits.max_concurrent_jobs,
                    )
                    return False, "max_concurrent_jobs_reached"

                # Add job to active set
                pipe.multi()
                pipe.sadd(key, job_id)
                pipe.expire(key, 86400)  # 24h TTL for safety
                await pipe.execute()

                logger.debug(
                    "slot_acquired",
                    tenant_id=tenant_id,
                    job_id=job_id,
                    active_count=current + 1,
                )
                return True, None

            except redis.WatchError:
                # Concurrent modification, retry
                return await self.try_acquire_slot(tenant_id, job_id)

    async def release_slot(
        self,
        tenant_id: str,
        job_id: str,
    ) -> None:
        """Release a job slot for tenant."""
        key = self._active_jobs_key(tenant_id)
        removed = await self.redis.srem(key, job_id)

        if removed:
            logger.debug(
                "slot_released",
                tenant_id=tenant_id,
                job_id=job_id,
            )
            # Check if any queued jobs can start
            await self._process_queued_jobs(tenant_id)

    async def get_active_count(self, tenant_id: str) -> int:
        """Get number of active jobs for tenant."""
        return await self.redis.scard(self._active_jobs_key(tenant_id))

    async def get_queued_count(self, tenant_id: str) -> int:
        """Get number of queued jobs for tenant."""
        return await self.redis.llen(self._queue_key(tenant_id))

    async def queue_job(
        self,
        tenant_id: str,
        job_id: str,
        task_data: dict,
    ) -> None:
        """Queue a job for later execution."""
        import json
        key = self._queue_key(tenant_id)
        await self.redis.rpush(key, json.dumps({
            "job_id": job_id,
            "task_data": task_data,
            "queued_at": datetime.utcnow().isoformat(),
        }))
        logger.info(
            "job_queued",
            tenant_id=tenant_id,
            job_id=job_id,
        )

    async def _process_queued_jobs(self, tenant_id: str) -> None:
        """Process any queued jobs that can now run."""
        from ingestion.tasks.ingest import process_document

        queue_key = self._queue_key(tenant_id)

        while True:
            # Try to acquire a slot
            # Peek at next job without removing
            job_data = await self.redis.lindex(queue_key, 0)
            if not job_data:
                break

            import json
            job = json.loads(job_data)
            job_id = job["job_id"]

            acquired, _ = await self.try_acquire_slot(tenant_id, job_id)
            if not acquired:
                break

            # Remove from queue and dispatch
            await self.redis.lpop(queue_key)

            # Re-dispatch the task
            process_document.apply_async(
                kwargs=job["task_data"],
                task_id=job_id,
            )

            logger.info(
                "queued_job_dispatched",
                tenant_id=tenant_id,
                job_id=job_id,
            )
```

### Celery Task Integration

```python
# services/ingestion/tasks/ingest.py

from celery import shared_task
from ingestion.rate_limiting.limiter import IngestionRateLimiter

rate_limiter = IngestionRateLimiter(redis_client)

@shared_task(
    bind=True,
    queue="ingestion_normal",
    max_retries=3,
)
async def process_document(
    self,
    document_id: str,
    tenant_id: str,
    source_type: str,
    **kwargs,
) -> dict:
    """
    Process document with rate limiting.

    Rate limiting is checked before processing begins.
    Slot is released on completion or failure.
    """
    job_id = self.request.id

    # Try to acquire slot
    acquired, reason = await rate_limiter.try_acquire_slot(tenant_id, job_id)

    if not acquired:
        limits = await rate_limiter.get_tenant_limits(tenant_id)

        if limits.hard_limit:
            # Reject with error
            logger.warning(
                "job_rejected_rate_limit",
                tenant_id=tenant_id,
                job_id=job_id,
            )
            raise RateLimitExceeded(f"Rate limit exceeded for tenant {tenant_id}")

        # Queue for later
        await rate_limiter.queue_job(
            tenant_id=tenant_id,
            job_id=job_id,
            task_data={
                "document_id": document_id,
                "tenant_id": tenant_id,
                "source_type": source_type,
                **kwargs,
            },
        )
        return {"status": "queued", "job_id": job_id}

    try:
        # Process the document
        result = await _process_document_impl(
            document_id=document_id,
            tenant_id=tenant_id,
            source_type=source_type,
            **kwargs,
        )
        return result
    finally:
        # Always release slot
        await rate_limiter.release_slot(tenant_id, job_id)


class RateLimitExceeded(Exception):
    """Raised when rate limit is exceeded in hard limit mode."""
    pass
```

```python
from celery import Celery

app = Celery("ingestion")

app.conf.task_queues = [
    Queue("ingestion_high", routing_key="ingestion.high"),

app.conf.task_routes = {
    "ingestion.tasks.ingest.process_document": {
        "queue": "ingestion_normal",  # Default, can be overridden
    },
}

# Worker configuration - process high priority first
app.conf.worker_prefetch_multiplier = 1

# Custom task routing based on tenant priority
def route_by_tenant_priority(name, args, kwargs, options, task=None, **kw):
    """Route tasks to appropriate queue based on tenant priority."""
    tenant_id = kwargs.get("tenant_id")
    if not tenant_id:
        return {"queue": "ingestion_normal"}

    # Get tenant limits (sync call for routing)
    import redis
    r = redis.Redis.from_url(REDIS_URL)
    priority = r.hget(f"ingestion:rate_limit:limits:{tenant_id}", "priority")

    if priority == b"high":
        return {"queue": "ingestion_high"}
    elif priority == b"low":
        return {"queue": "ingestion_low"}

    return {"queue": "ingestion_normal"}

app.conf.task_routes = [route_by_tenant_priority]
```

```python
# services/ingestion/api/admin_routes.py

from pydantic import BaseModel

class TenantLimitsRequest(BaseModel):
    max_concurrent_jobs: int = 10
    priority: str = "normal"
    hard_limit: bool = False

class TenantJobStats(BaseModel):
    tenant_id: str
    active_jobs: int
    queued_jobs: int
    max_concurrent: int
    priority: str

@router.get("/admin/tenants/{tenant_id}/rate-limits")
async def get_tenant_rate_limits(
    tenant_id: str,
    rate_limiter: IngestionRateLimiter = Depends(get_rate_limiter),
    current_user: User = Depends(require_admin),
) -> TenantJobStats:
    """Get rate limit configuration and current job counts."""
    limits = await rate_limiter.get_tenant_limits(tenant_id)
    active = await rate_limiter.get_active_count(tenant_id)
    queued = await rate_limiter.get_queued_count(tenant_id)

    return TenantJobStats(
        tenant_id=tenant_id,
        active_jobs=active,
        queued_jobs=queued,
        max_concurrent=limits.max_concurrent_jobs,
        priority=limits.priority,
    )

@router.put("/admin/tenants/{tenant_id}/rate-limits")
async def update_tenant_rate_limits(
    tenant_id: str,
    request: TenantLimitsRequest,
    rate_limiter: IngestionRateLimiter = Depends(get_rate_limiter),
    current_user: User = Depends(require_admin),
) -> dict:
    """Update rate limit configuration for tenant."""
    limits = TenantLimits(
        max_concurrent_jobs=request.max_concurrent_jobs,
        priority=request.priority,
        hard_limit=request.hard_limit,
    )
    await rate_limiter.set_tenant_limits(tenant_id, limits)

    return {"status": "updated", "tenant_id": tenant_id, "limits": request.dict()}

@router.get("/admin/rate-limits/overview")
async def get_rate_limits_overview(
    rate_limiter: IngestionRateLimiter = Depends(get_rate_limiter),
    current_user: User = Depends(require_admin),
) -> dict:
    """Get overview of all tenant job activity."""
    # Get all active tenant keys
    keys = await rate_limiter.redis.keys("ingestion:rate_limit:active:*")

    tenants = []
    for key in keys:
        tenant_id = key.decode().split(":")[-1]
        stats = await get_tenant_rate_limits(tenant_id, rate_limiter, current_user)
        tenants.append(stats)

    return {
        "total_active_tenants": len(tenants),
        "tenants": tenants,
    }

@router.post("/admin/tenants/{tenant_id}/clear-queue")
async def clear_tenant_queue(
    tenant_id: str,
    rate_limiter: IngestionRateLimiter = Depends(get_rate_limiter),
    current_user: User = Depends(require_admin),
) -> dict:
    """Clear all queued jobs for tenant (emergency use)."""
    queue_key = rate_limiter._queue_key(tenant_id)
    count = await rate_limiter.redis.llen(queue_key)
    await rate_limiter.redis.delete(queue_key)

    logger.warning(
        "tenant_queue_cleared",
        tenant_id=tenant_id,
        cleared_count=count,
        admin_user=current_user.id,
    )

    return {"status": "cleared", "jobs_removed": count}
```
