# US-10.7.1: Inter-Service Authentication

> **Initiative:** 7 - Security & Production Hardening
> **Priority:** P0
> **Effort:** Medium (2-3 days)
> **Status:** Draft

## Summary

Implement mutual TLS (mTLS) or JWT-based service-to-service authentication to prevent unauthorized internal communication.

## Problem Statement

**Current state**: Services communicate without authentication:
- Any network-accessible client can call internal APIs
- No verification of service identity
- Lateral movement risk if one service is compromised
- No audit trail of service-to-service calls

**Impact**:
- Security vulnerability in production
- Compliance gaps (SOC2, ISO27001)
- Difficult to detect unauthorized access
- No defense in depth

## User Story

**As a** security engineer
**I want** authenticated service-to-service communication
**So that** only authorized services can access internal APIs

## Acceptance Criteria

### AC-1: Service Identity
- [ ] Each service has a unique identity/credential
- [ ] Credentials are rotated regularly
- [ ] Compromised credentials can be revoked

### AC-2: Request Authentication
- [ ] All inter-service requests include authentication
- [ ] Receiving service validates authentication
- [ ] Invalid requests are rejected with 401

### AC-3: Authorization
- [ ] Services can only call permitted endpoints
- [ ] Authorization matrix is configurable
- [ ] Unauthorized calls return 403

### AC-4: Audit Logging
- [ ] All service-to-service calls are logged
- [ ] Log includes caller identity and target
- [ ] Failed auth attempts logged separately

### JWT-Based Approach

```python
# services/shared/auth/service_auth.py

import jwt
import time
from dataclasses import dataclass
from functools import lru_cache
from typing import Optional
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric import rsa

@dataclass
class ServiceIdentity:
    """Service identity with signing key."""
    service_name: str
    private_key: rsa.RSAPrivateKey
    public_key: rsa.RSAPublicKey
    allowed_targets: list[str]

class ServiceAuthClient:
    """Client for generating service-to-service auth tokens."""

    def __init__(
        self,
        identity: ServiceIdentity,
        token_ttl: int = 300,  # 5 minutes
    ):
        self.identity = identity
        self.token_ttl = token_ttl
        self._token_cache: tuple[str, float] | None = None

    def get_auth_headers(self, target_service: str) -> dict[str, str]:
        """Get authorization headers for target service."""
        token = self._get_or_create_token(target_service)
        return {"Authorization": f"Bearer {token}"}

    def _get_or_create_token(self, target_service: str) -> str:
        """Get cached token or create new one."""
        now = time.time()

        # Check cache
        if self._token_cache:
            token, expiry = self._token_cache
            if expiry > now + 30:  # 30s buffer
                return token

        # Create new token
        token = self._create_token(target_service)
        self._token_cache = (token, now + self.token_ttl)
        return token

    def _create_token(self, target_service: str) -> str:
        """Create signed JWT for service-to-service auth."""
        now = int(time.time())

        payload = {
            "iss": self.identity.service_name,
            "sub": self.identity.service_name,
            "aud": target_service,
            "iat": now,
            "exp": now + self.token_ttl,
            "jti": f"{self.identity.service_name}-{now}",
        }

        private_key_pem = self.identity.private_key.private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PrivateFormat.PKCS8,
            encryption_algorithm=serialization.NoEncryption(),
        )

        return jwt.encode(payload, private_key_pem, algorithm="RS256")


class ServiceAuthValidator:
    """Validator for incoming service-to-service requests."""

    def __init__(
        self,
        service_name: str,
        trusted_services: dict[str, rsa.RSAPublicKey],
        authorization_matrix: dict[str, list[str]],
    ):
        self.service_name = service_name
        self.trusted_services = trusted_services
        self.authorization_matrix = authorization_matrix

    def validate_request(
        self,
        token: str,
        endpoint: str,
    ) -> tuple[bool, str, Optional[str]]:
        """
        Validate service-to-service request.

        Returns: (is_valid, caller_service, error_message)
        """
        try:
            # Decode header to get issuer
            unverified = jwt.decode(
                token,
                options={"verify_signature": False},
            )
            issuer = unverified.get("iss")

            if issuer not in self.trusted_services:
                return False, "", f"Unknown issuer: {issuer}"

            # Verify with issuer's public key
            public_key = self.trusted_services[issuer]
            public_key_pem = public_key.public_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PublicFormat.SubjectPublicKeyInfo,
            )

            payload = jwt.decode(
                token,
                public_key_pem,
                algorithms=["RS256"],
                audience=self.service_name,
            )

            caller = payload["sub"]

            # Check authorization
            allowed_endpoints = self.authorization_matrix.get(caller, [])
            if not self._is_endpoint_allowed(endpoint, allowed_endpoints):
                return False, caller, f"Service {caller} not authorized for {endpoint}"

            return True, caller, None

        except jwt.ExpiredSignatureError:
            return False, "", "Token expired"
        except jwt.InvalidAudienceError:
            return False, "", "Invalid audience"
        except jwt.InvalidTokenError as e:
            return False, "", f"Invalid token: {e}"

    def _is_endpoint_allowed(
        self,
        endpoint: str,
        allowed_patterns: list[str],
    ) -> bool:
        """Check if endpoint matches allowed patterns."""
        import fnmatch
        return any(fnmatch.fnmatch(endpoint, pattern) for pattern in allowed_patterns)
```

### FastAPI Middleware

```python
# services/shared/auth/middleware.py

from fastapi import Request, HTTPException
from starlette.middleware.base import BaseHTTPMiddleware
import structlog

logger = structlog.get_logger()

class ServiceAuthMiddleware(BaseHTTPMiddleware):
    """Middleware for validating service-to-service authentication."""

    def __init__(
        self,
        app,
        validator: ServiceAuthValidator,
        exclude_paths: list[str] | None = None,
    ):
        super().__init__(app)
        self.validator = validator
        self.exclude_paths = exclude_paths or ["/health", "/metrics", "/docs", "/openapi.json"]

    async def dispatch(self, request: Request, call_next):
        # Skip excluded paths
        if any(request.url.path.startswith(p) for p in self.exclude_paths):
            return await call_next(request)

        # Check for external API requests (user auth handled separately)
        if request.url.path.startswith("/api/v1/"):
            # User-facing endpoints use different auth
            return await call_next(request)

        # Validate service auth for internal endpoints
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith("Bearer "):
            logger.warning(
                "missing_service_auth",
                path=request.url.path,
                client_host=request.client.host if request.client else "unknown",
            )
            raise HTTPException(status_code=401, detail="Service authentication required")

        token = auth_header.replace("Bearer ", "")
        is_valid, caller, error = self.validator.validate_request(
            token,
            request.url.path,
        )

        if not is_valid:
            logger.warning(
                "service_auth_failed",
                path=request.url.path,
                caller=caller or "unknown",
                error=error,
            )
            if "not authorized" in (error or ""):
                raise HTTPException(status_code=403, detail=error)
            raise HTTPException(status_code=401, detail=error)

        # Add caller to request state for logging
        request.state.caller_service = caller

        logger.debug(
            "service_auth_success",
            path=request.url.path,
            caller=caller,
        )

        return await call_next(request)
```

```python
# services/shared/http/authenticated_client.py

import httpx
from typing import Any

class AuthenticatedServiceClient:
    """HTTP client with automatic service authentication."""

    def __init__(
        self,
        base_url: str,
        target_service: str,
        auth_client: ServiceAuthClient,
        timeout: float = 30.0,
    ):
        self.base_url = base_url.rstrip("/")
        self.target_service = target_service
        self.auth_client = auth_client
        self.timeout = timeout

    async def request(
        self,
        method: str,
        path: str,
        **kwargs,
    ) -> httpx.Response:
        """Make authenticated request to target service."""
        headers = kwargs.pop("headers", {})
        auth_headers = self.auth_client.get_auth_headers(self.target_service)
        headers.update(auth_headers)

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.request(
                method,
                f"{self.base_url}{path}",
                headers=headers,
                **kwargs,
            )
            return response

    async def get(self, path: str, **kwargs) -> httpx.Response:
        return await self.request("GET", path, **kwargs)

    async def post(self, path: str, **kwargs) -> httpx.Response:
        return await self.request("POST", path, **kwargs)


# Usage in orchestrator
class RetrievalClient:
    """Client for calling retrieval service."""

    def __init__(self, auth_client: ServiceAuthClient):
        self.client = AuthenticatedServiceClient(
            base_url="http://retrieval:8002",
            target_service="retrieval",
            auth_client=auth_client,
        )

    async def search(self, query: str, tenant_id: str, **kwargs) -> dict:
        response = await self.client.post(
            "/internal/search",
            json={
                "query": query,
                "tenant_id": tenant_id,
                **kwargs,
            },
        )
        response.raise_for_status()
        return response.json()
```

### Key Management

```python
# services/shared/auth/key_manager.py

import os
from pathlib import Path
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.backends import default_backend

class KeyManager:
    """Manage service authentication keys."""

    def __init__(self, keys_dir: str = "/etc/service-auth"):
        self.keys_dir = Path(keys_dir)

    def load_private_key(self, service_name: str) -> rsa.RSAPrivateKey:
        """Load service's private key."""
        key_path = self.keys_dir / f"{service_name}.key"

        with open(key_path, "rb") as f:
            return serialization.load_pem_private_key(
                f.read(),
                password=None,
                backend=default_backend(),
            )

    def load_public_key(self, service_name: str) -> rsa.RSAPublicKey:
        """Load service's public key."""
        key_path = self.keys_dir / f"{service_name}.pub"

        with open(key_path, "rb") as f:
            return serialization.load_pem_public_key(
                f.read(),
                backend=default_backend(),
            )

    def load_trusted_services(
        self,
        service_names: list[str],
    ) -> dict[str, rsa.RSAPublicKey]:
        """Load public keys for trusted services."""
        return {
            name: self.load_public_key(name)
            for name in service_names
        }

    @staticmethod
    def generate_keypair() -> tuple[rsa.RSAPrivateKey, rsa.RSAPublicKey]:
        """Generate new RSA keypair for a service."""
        private_key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=2048,
            backend=default_backend(),
        )
        return private_key, private_key.public_key()
```
