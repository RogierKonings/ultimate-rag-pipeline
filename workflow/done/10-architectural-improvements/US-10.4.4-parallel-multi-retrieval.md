# US-10.4.4: Parallel Multi-Retrieval

> **Initiative:** 4 - Modern RAG Techniques
> **Priority:** P2 (Optional)
> **Effort:** Medium (2-3 days)
> **Status:** Draft

## Summary

Implement parallel retrieval for decomposed sub-questions, aggregating and deduplicating context for comprehensive answers.

## Problem Statement

**Context**: US-10.4.3 decomposes complex queries into sub-questions, but:
- No mechanism to retrieve for multiple sub-questions
- Sequential retrieval would be too slow
- Results need aggregation and deduplication

## User Story

**As a** user with multi-hop queries
**I want** the system to retrieve context for each sub-question
**So that** all aspects of my question are addressed

## Acceptance Criteria

### AC-1: Parallel Retrieval
- [ ] New `multi_retrieval` node after decomposition
- [ ] Parallel retrieval for all sub-questions (asyncio.gather)
- [ ] Respects retrieval service rate limits

- [ ] Merge results from all sub-queries
- [ ] Deduplicate based on chunk_id

### AC-3: Relevance Scoring
- [ ] Weight results by sub-question relevance
- [ ] Top-K selection across all results
- [ ] Configurable total context limit

### AC-4: Performance
- [ ] Latency target: < 500ms overhead for parallelism
- [ ] Efficient deduplication

### Multi-Retrieval Node

```python
# services/orchestrator/workflow/nodes/multi_retrieval.py

import asyncio
from dataclasses import dataclass

@dataclass
class SubQueryResult:
    """Result from a single sub-question retrieval."""
    sub_question: str
    documents: list[Document]
    latency_ms: float

@dataclass
class AggregatedContext:
    """Aggregated and deduplicated context."""
    documents: list[Document]
    sub_question_mapping: dict[str, list[str]]  # doc_id -> sub_questions
    total_retrieved: int
    deduplicated_count: int

async def multi_retrieval_node(state: RAGState) -> RAGState:
    """
    Retrieve context for all sub-questions in parallel.

    Aggregates and deduplicates results.
    """
    import time
    start = time.time()

    sub_questions = state.get("sub_questions", [state["query"]])
    user_context = state["user_context"]
    retrieval_client = state["retrieval_client"]
    options = state.get("retrieval_options", {})

    # Parallel retrieval for all sub-questions
    tasks = [
        _retrieve_for_sub_question(
            retrieval_client,
            sq,
            user_context,
            options,
        )
        for sq in sub_questions
    ]

    results = await asyncio.gather(*tasks, return_exceptions=True)

    # Process results
    sub_query_results = []
    for sq, result in zip(sub_questions, results):
        if isinstance(result, Exception):
            logger.warning("sub_query_retrieval_failed", sub_question=sq, error=str(result))
            sub_query_results.append(SubQueryResult(sq, [], 0))
        else:
            sub_query_results.append(result)

    # Aggregate and deduplicate
    aggregated = _aggregate_results(
        sub_query_results,
        max_documents=options.get("max_total_documents", 20),
    )

    state["documents"] = aggregated.documents
    state["sub_question_mapping"] = aggregated.sub_question_mapping
    state["retrieval_stats"] = {
        "sub_questions": len(sub_questions),
        "total_retrieved": aggregated.total_retrieved,
        "after_dedup": aggregated.deduplicated_count,
        "latency_ms": (time.time() - start) * 1000,
    }

    logger.info(
        "multi_retrieval_complete",
        sub_questions=len(sub_questions),
        total_documents=len(aggregated.documents),
        dedup_removed=aggregated.total_retrieved - aggregated.deduplicated_count,
    )

    return state

async def _retrieve_for_sub_question(
    client,
    sub_question: str,
    user_context,
    options: dict,
) -> SubQueryResult:
    """Retrieve documents for a single sub-question."""
    import time
    start = time.time()

    response = await client.search(
        query=sub_question,
        user_context=user_context,
        top_k=options.get("sub_question_top_k", 10),
    )

    return SubQueryResult(
        sub_question=sub_question,
        documents=response["results"],
        latency_ms=(time.time() - start) * 1000,
    )

def _aggregate_results(
    results: list[SubQueryResult],
    max_documents: int = 20,
) -> AggregatedContext:
    """Aggregate and deduplicate results from sub-questions."""
    seen_chunks: dict[str, Document] = {}
    sub_question_mapping: dict[str, list[str]] = {}
    total_retrieved = 0

    for result in results:
        total_retrieved += len(result.documents)

        for doc in result.documents:
            chunk_id = doc.chunk_id

            if chunk_id not in seen_chunks:
                seen_chunks[chunk_id] = doc
                sub_question_mapping[chunk_id] = [result.sub_question]
            else:
                # Document seen before, add sub-question attribution
                sub_question_mapping[chunk_id].append(result.sub_question)
                # Boost score for documents relevant to multiple sub-questions
                seen_chunks[chunk_id].relevance_score += doc.relevance_score * 0.5

    # Sort by boosted score and take top K
    sorted_docs = sorted(
        seen_chunks.values(),
        key=lambda d: d.relevance_score,
        reverse=True,
    )[:max_documents]

    return AggregatedContext(
        documents=sorted_docs,
        sub_question_mapping={
            d.chunk_id: sub_question_mapping[d.chunk_id]
            for d in sorted_docs
        },
        total_retrieved=total_retrieved,
        deduplicated_count=len(sorted_docs),
    )
```

### Enhanced Prompt Building

```python
# services/orchestrator/workflow/nodes/prompt_building.py

def _format_multi_hop_context(
    documents: list[Document],
    sub_question_mapping: dict[str, list[str]],
    original_query: str,
    sub_questions: list[str],
) -> str:
    """Format context with sub-question attribution."""
    context_parts = []

    # Group documents by sub-question relevance
    for i, sq in enumerate(sub_questions, 1):
        sq_docs = [
            d for d in documents
            if sq in sub_question_mapping.get(d.chunk_id, [])
        ]

        if sq_docs:
            context_parts.append(f"\n### Context for: {sq}\n")
            for doc in sq_docs[:5]:  # Limit per sub-question
                context_parts.append(f"[{doc.chunk_id[:8]}] {doc.content}\n")

    return "\n".join(context_parts)
```

```python
async def test_deduplication_removes_duplicates():
    """Same chunk from multiple sub-questions should be deduplicated."""
    results = [
        SubQueryResult("q1", [Document(chunk_id="a", relevance_score=0.9)], 100),
        SubQueryResult("q2", [Document(chunk_id="a", relevance_score=0.8)], 100),
    ]

    aggregated = _aggregate_results(results)

    assert len(aggregated.documents) == 1
    assert aggregated.documents[0].chunk_id == "a"
    # Score should be boosted
    assert aggregated.documents[0].relevance_score > 0.9
```
