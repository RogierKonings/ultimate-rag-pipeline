## Summary

Cache complete RAG responses to serve instant answers for repeated questions, reducing LLM costs.

## User Story

**As a** user asking common questions
**I want** instant responses for previously asked questions
**So that** I get faster answers and the system saves resources

- [x] Cache invalidation when source documents change

## Technical Design

```python
# services/orchestrator/cache/answer_cache.py

import hashlib
import json
from redis.asyncio import Redis
from pydantic import BaseModel

class CachedAnswer(BaseModel):
    """Cached RAG answer."""
    response: str
    citations: list[dict]
    model_used: str
    cached_at: str
    retrieval_mode: str

class AnswerCache:
    """Cache for complete RAG answers."""

    def __init__(self, redis: Redis, default_ttl: int = 3600):
        self.redis = redis
        self.default_ttl = default_ttl
        self.key_prefix = "rag:answer_cache"

    def _build_key(
        self,
        tenant_id: str,
        query: str,
        config_hash: str,
    ) -> str:
        """Build cache key from components."""
        normalized_query = self._normalize_query(query)
        query_hash = hashlib.sha256(normalized_query.encode()).hexdigest()[:16]
        return f"{self.key_prefix}:{tenant_id}:{query_hash}:{config_hash}"

    def _normalize_query(self, query: str) -> str:
        """Normalize query for cache key."""
        return query.lower().strip()

    async def get(
        self,
        tenant_id: str,
        query: str,
        config_hash: str,
    ) -> CachedAnswer | None:
        """Get cached answer if available."""
        key = self._build_key(tenant_id, query, config_hash)
        data = await self.redis.get(key)

        if data:
            return CachedAnswer(**json.loads(data))
        return None

    async def set(
        self,
        tenant_id: str,
        query: str,
        config_hash: str,
        answer: CachedAnswer,
        ttl: int | None = None,
    ) -> None:
        """Cache an answer."""
        key = self._build_key(tenant_id, query, config_hash)
        await self.redis.setex(
            key,
            ttl or self.default_ttl,
            answer.json(),
        )

    async def invalidate_for_document(
        self,
        tenant_id: str,
        document_id: str,
    ) -> int:
        """Invalidate cache entries that used a document."""
        # Pattern-based invalidation
        pattern = f"{self.key_prefix}:{tenant_id}:*"
        count = 0
        async for key in self.redis.scan_iter(pattern):
            # Check if document was used (requires storing doc refs)
            await self.redis.delete(key)
            count += 1
        return count
```
