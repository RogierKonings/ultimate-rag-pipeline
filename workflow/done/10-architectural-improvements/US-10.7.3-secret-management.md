# US-10.7.3: Secret Management

> **Initiative:** 7 - Security & Production Hardening
> **Priority:** P0
> **Effort:** Medium (2-3 days)
> **Status:** Draft

## Summary

Implement centralized secret management using HashiCorp Vault or Kubernetes External Secrets to eliminate hardcoded credentials.

## Problem Statement

**Current state**: Secrets are managed ad-hoc:
- Credentials in environment variables
- Secrets in Kubernetes Secrets (base64 encoded, not encrypted)
- No secret rotation mechanism
- No audit trail for secret access
- Manual secret distribution

**Impact**:
- Security risk from static secrets
- Difficult to rotate compromised credentials
- No visibility into secret usage
- Compliance gaps

## User Story

**As a** security engineer
**I want** centralized secret management with rotation
**So that** credentials are secure and auditable

## Acceptance Criteria

### AC-1: Centralized Secret Store
- [ ] All secrets stored in Vault or external secret store
- [ ] No secrets in ConfigMaps or plain Secrets
- [ ] Secrets encrypted at rest

### AC-2: Dynamic Secret Injection
- [ ] Secrets injected at runtime
- [ ] No secrets in container images
- [ ] Support for sidecar and init container patterns

### AC-3: Secret Rotation
- [ ] Database credentials rotatable without restart
- [ ] API keys refreshable automatically
- [ ] Grace period for old credentials during rotation

### AC-4: Audit Logging
- [ ] All secret access logged
- [ ] Who accessed what and when
- [ ] Alert on unusual access patterns

### Vault Client Implementation

```python
# services/shared/secrets/vault_client.py

import os
import asyncio
from typing import Any, Optional
from dataclasses import dataclass
import hvac
import structlog

logger = structlog.get_logger()

@dataclass
class VaultConfig:
    """Vault connection configuration."""
    address: str
    namespace: Optional[str] = None
    auth_method: str = "kubernetes"
    role: Optional[str] = None
    token: Optional[str] = None
    mount_point: str = "secret"

class VaultClient:
    """Client for HashiCorp Vault secret management."""

    def __init__(self, config: VaultConfig):
        self.config = config
        self._client: Optional[hvac.Client] = None
        self._token_expires: float = 0

    async def initialize(self):
        """Initialize Vault client with authentication."""
        self._client = hvac.Client(
            url=self.config.address,
            namespace=self.config.namespace,
        )

        if self.config.auth_method == "kubernetes":
            await self._kubernetes_auth()
        elif self.config.auth_method == "token":
            self._client.token = self.config.token
        else:
            raise ValueError(f"Unknown auth method: {self.config.auth_method}")

        logger.info("vault_initialized", address=self.config.address)

    async def _kubernetes_auth(self):
        """Authenticate using Kubernetes service account."""
        # Read service account token
        sa_token_path = "/var/run/secrets/kubernetes.io/serviceaccount/token"
        with open(sa_token_path, "r") as f:
            jwt = f.read()

        # Authenticate with Vault
        response = self._client.auth.kubernetes.login(
            role=self.config.role,
            jwt=jwt,
        )

        self._client.token = response["auth"]["client_token"]
        self._token_expires = asyncio.get_event_loop().time() + response["auth"]["lease_duration"]

        logger.info("vault_kubernetes_auth_success", role=self.config.role)

    async def _ensure_authenticated(self):
        """Ensure client is authenticated, refresh if needed."""
        if asyncio.get_event_loop().time() > self._token_expires - 60:
            await self._kubernetes_auth()

    async def get_secret(self, path: str, key: Optional[str] = None) -> Any:
        """
        Get secret from Vault KV v2.

        Args:
            path: Secret path (e.g., "rag-pipeline/database")
            key: Optional specific key within the secret

        Returns:
            Secret value or dict of all values
        """
        await self._ensure_authenticated()

        try:
            response = self._client.secrets.kv.v2.read_secret_version(
                path=path,
                mount_point=self.config.mount_point,
            )

            data = response["data"]["data"]

            if key:
                return data.get(key)
            return data

        except hvac.exceptions.InvalidPath:
            logger.warning("vault_secret_not_found", path=path)
            return None

    async def get_database_credentials(
        self,
        database: str,
        role: str,
    ) -> dict[str, str]:
        """
        Get dynamic database credentials from Vault database engine.

        Args:
            database: Database mount name
            role: Database role name

        Returns:
            Dict with username and password
        """
        await self._ensure_authenticated()

        response = self._client.secrets.database.generate_credentials(
            name=role,
            mount_point=database,
        )

        return {
            "username": response["data"]["username"],
            "password": response["data"]["password"],
            "lease_id": response["lease_id"],
            "lease_duration": response["lease_duration"],
        }

    async def renew_lease(self, lease_id: str) -> int:
        """Renew a lease and return new duration."""
        await self._ensure_authenticated()

        response = self._client.sys.renew_lease(lease_id=lease_id)
        return response["lease_duration"]

    async def revoke_lease(self, lease_id: str):
        """Revoke a lease (e.g., on shutdown)."""
        await self._ensure_authenticated()

        self._client.sys.revoke_lease(lease_id=lease_id)
        logger.info("vault_lease_revoked", lease_id=lease_id)
```

### Secret Provider Interface

```python
# services/shared/secrets/provider.py

from abc import ABC, abstractmethod
from typing import Any, Optional
from dataclasses import dataclass
import os

@dataclass
class SecretValue:
    """Wrapper for secret value with metadata."""
    value: Any
    version: Optional[str] = None
    expires_at: Optional[float] = None
    lease_id: Optional[str] = None

class SecretProvider(ABC):
    """Abstract interface for secret providers."""

    @abstractmethod
    async def get(self, key: str) -> SecretValue:
        """Get a secret by key."""
        pass

    @abstractmethod
    async def refresh(self, key: str) -> SecretValue:
        """Refresh a secret (get latest version)."""
        pass

class VaultSecretProvider(SecretProvider):
    """Vault-backed secret provider."""

    def __init__(self, vault_client: VaultClient, base_path: str):
        self.vault = vault_client
        self.base_path = base_path
        self._cache: dict[str, SecretValue] = {}

    async def get(self, key: str) -> SecretValue:
        """Get secret, using cache if available."""
        if key in self._cache:
            cached = self._cache[key]
            if cached.expires_at is None or cached.expires_at > time.time():
                return cached

        return await self.refresh(key)

    async def refresh(self, key: str) -> SecretValue:
        """Refresh secret from Vault."""
        path = f"{self.base_path}/{key}"
        value = await self.vault.get_secret(path)

        secret = SecretValue(value=value)
        self._cache[key] = secret

        return secret

class EnvironmentSecretProvider(SecretProvider):
    """Environment variable secret provider (for local dev)."""

    def __init__(self, prefix: str = ""):
        self.prefix = prefix

    async def get(self, key: str) -> SecretValue:
        """Get secret from environment."""
        env_key = f"{self.prefix}{key}".upper().replace("-", "_").replace("/", "_")
        value = os.getenv(env_key)
        return SecretValue(value=value)

    async def refresh(self, key: str) -> SecretValue:
        """Environment secrets don't refresh."""
        return await self.get(key)


def create_secret_provider() -> SecretProvider:
    """Factory function to create appropriate secret provider."""
    if os.getenv("VAULT_ADDR"):
        from shared.secrets.vault_client import VaultClient, VaultConfig

        config = VaultConfig(
            address=os.getenv("VAULT_ADDR"),
            namespace=os.getenv("VAULT_NAMESPACE"),
            auth_method=os.getenv("VAULT_AUTH_METHOD", "kubernetes"),
            role=os.getenv("VAULT_ROLE"),
            mount_point=os.getenv("VAULT_MOUNT_POINT", "secret"),
        )

        vault = VaultClient(config)
        return VaultSecretProvider(vault, os.getenv("VAULT_SECRET_PATH", "rag-pipeline"))

    # Fallback to environment for local development
    return EnvironmentSecretProvider(prefix="RAG_")
```

### Dynamic Database Credential Manager

```python
# services/shared/secrets/credential_manager.py

import asyncio
from typing import Optional, Callable
from dataclasses import dataclass
import structlog

logger = structlog.get_logger()

@dataclass
class DatabaseCredentials:
    """Dynamic database credentials."""
    username: str
    password: str
    lease_id: str
    expires_at: float

class DynamicCredentialManager:
    """
    Manages dynamic database credentials with automatic renewal.

    Handles credential rotation without service restart by:
    1. Getting initial credentials from Vault
    2. Scheduling renewal before expiry
    3. Notifying connection pools of credential changes
    """

    def __init__(
        self,
        vault_client: VaultClient,
        database: str,
        role: str,
        renewal_buffer: int = 300,  # 5 minutes before expiry
    ):
        self.vault = vault_client
        self.database = database
        self.role = role
        self.renewal_buffer = renewal_buffer
        self._credentials: Optional[DatabaseCredentials] = None
        self._renewal_task: Optional[asyncio.Task] = None
        self._on_credential_change: list[Callable] = []

    def on_credential_change(self, callback: Callable[[str, str], None]):
        """Register callback for credential changes."""
        self._on_credential_change.append(callback)

    async def start(self):
        """Start credential management."""
        await self._fetch_credentials()
        self._schedule_renewal()

    async def stop(self):
        """Stop credential management and revoke lease."""
        if self._renewal_task:
            self._renewal_task.cancel()

        if self._credentials:
            await self.vault.revoke_lease(self._credentials.lease_id)

    async def get_credentials(self) -> tuple[str, str]:
        """Get current credentials."""
        if not self._credentials:
            await self._fetch_credentials()

        return self._credentials.username, self._credentials.password

    async def _fetch_credentials(self):
        """Fetch new credentials from Vault."""
        creds = await self.vault.get_database_credentials(
            self.database,
            self.role,
        )

        self._credentials = DatabaseCredentials(
            username=creds["username"],
            password=creds["password"],
            lease_id=creds["lease_id"],
            expires_at=asyncio.get_event_loop().time() + creds["lease_duration"],
        )

        logger.info(
            "database_credentials_fetched",
            database=self.database,
            role=self.role,
            expires_in=creds["lease_duration"],
        )

        # Notify listeners
        for callback in self._on_credential_change:
            try:
                callback(self._credentials.username, self._credentials.password)
            except Exception as e:
                logger.error("credential_change_callback_failed", error=str(e))

    def _schedule_renewal(self):
        """Schedule credential renewal."""
        if not self._credentials:
            return

        delay = self._credentials.expires_at - asyncio.get_event_loop().time() - self.renewal_buffer

        if delay > 0:
            self._renewal_task = asyncio.create_task(self._renewal_loop(delay))

    async def _renewal_loop(self, initial_delay: float):
        """Loop to renew credentials periodically."""
        await asyncio.sleep(initial_delay)

        while True:
            try:
                # Try to renew existing lease
                new_duration = await self.vault.renew_lease(self._credentials.lease_id)
                self._credentials.expires_at = asyncio.get_event_loop().time() + new_duration

                logger.info(
                    "database_credential_lease_renewed",
                    database=self.database,
                    new_duration=new_duration,
                )

                # Sleep until next renewal
                await asyncio.sleep(new_duration - self.renewal_buffer)

            except Exception as e:
                logger.warning(
                    "database_credential_renewal_failed",
                    error=str(e),
                    message="Fetching new credentials",
                )

                # Fetch new credentials
                await self._fetch_credentials()
                self._schedule_renewal()
                break
```
