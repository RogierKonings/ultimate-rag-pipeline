# US-10.4.3: Extended Routing with Multi-Hop Strategy

> **Initiative:** 4 - Modern RAG Techniques
> **Priority:** P2 (Optional)
> **Effort:** Medium (3-4 days)
> **Status:** Draft

## Summary

Extend the query routing system to detect complex multi-hop queries and route them to a decomposition pathway for comprehensive answers.

## Problem Statement

**Current state**: The router in [routing.py](services/orchestrator/workflow/nodes/routing.py) classifies queries but only supports simple vs complex strategies:
- Complex queries requiring multiple facts are handled as single queries
- Multi-hop reasoning not supported
- Comparison questions get incomplete answers

**Impact**:
- Complex questions get partial answers
- User satisfaction lower for multi-faceted queries
- Cannot compete with advanced RAG systems

## User Story

**As a** user asking complex questions
**I want** the system to break down my query into sub-questions
**So that** I get comprehensive answers for complex topics

```python
# services/orchestrator/workflow/routing/strategies.py

from enum import Enum

class RoutingStrategy(str, Enum):
    SIMPLE = "simple"           # Direct retrieval + generation
    COMPLEX = "complex"         # Retrieval with reranking
    MULTI_HOP = "multi_hop"     # Decompose into sub-questions
    AGGREGATION = "aggregation" # Collect and summarize multiple sources
    COMPARISON = "comparison"   # Compare multiple entities
    NO_RETRIEVAL = "no_retrieval"  # Direct LLM (chitchat, etc.)

class MultiHopIndicators:
    """Patterns indicating multi-hop queries."""

    COMPARISON_PATTERNS = [
        r"compare\s+",
        r"difference\s+between",
        r"vs\.?\s+",
        r"versus\s+",
        r"better\s+than",
        r"how\s+does\s+.+\s+differ",
        r"similarities?\s+and\s+differences?",
    ]

    AGGREGATION_PATTERNS = [
        r"list\s+all",
        r"what\s+are\s+all",
        r"summarize\s+",
        r"overview\s+of",
        r"everything\s+about",
    ]

    SEQUENTIAL_PATTERNS = [
        r"first.+then",
        r"step\s+by\s+step",
        r"what\s+happens\s+.+\s+and\s+then",
        r"after\s+that",
    ]

    MULTI_ENTITY_PATTERNS = [
        r"both\s+.+\s+and\s+",
        r"between\s+.+\s+and\s+",
        r"(\w+),\s*(\w+),?\s*and\s+(\w+)",  # Lists
    ]
```

### Extended Query Router

```python
# services/orchestrator/workflow/routing/router.py

import re
from orchestrator.workflow.routing.strategies import (
    RoutingStrategy,
    MultiHopIndicators,
)

class ExtendedQueryRouter:
    """Router with multi-hop detection."""

    def __init__(self, llm_client, complexity_scorer):
        self.llm = llm_client
        self.scorer = complexity_scorer

    async def route(self, query: str, context: dict) -> RoutingResult:
        """Determine routing strategy for query."""
        # Check for multi-hop patterns first
        multi_hop_type = self._detect_multi_hop_type(query)

        if multi_hop_type == "comparison":
            return RoutingResult(
                strategy=RoutingStrategy.COMPARISON,
                confidence=0.9,
                metadata={"multi_hop_type": "comparison"},
            )

        if multi_hop_type == "aggregation":
            return RoutingResult(
                strategy=RoutingStrategy.AGGREGATION,
                confidence=0.9,
                metadata={"multi_hop_type": "aggregation"},
            )

        if multi_hop_type == "sequential":
            return RoutingResult(
                strategy=RoutingStrategy.MULTI_HOP,
                confidence=0.9,
                metadata={"multi_hop_type": "sequential"},
            )

        # Fall back to complexity scoring
        complexity = await self.scorer.score(query)

        if complexity.score > 0.7:
            return RoutingResult(
                strategy=RoutingStrategy.COMPLEX,
                confidence=complexity.confidence,
            )

        return RoutingResult(
            strategy=RoutingStrategy.SIMPLE,
            confidence=complexity.confidence,
        )

    def _detect_multi_hop_type(self, query: str) -> str | None:
        """Detect type of multi-hop query."""
        query_lower = query.lower()

        for pattern in MultiHopIndicators.COMPARISON_PATTERNS:
            if re.search(pattern, query_lower):
                return "comparison"

        for pattern in MultiHopIndicators.AGGREGATION_PATTERNS:
            if re.search(pattern, query_lower):
                return "aggregation"

        for pattern in MultiHopIndicators.SEQUENTIAL_PATTERNS:
            if re.search(pattern, query_lower):
                return "sequential"

        return None
```

### Decomposition Node

```python
# services/orchestrator/workflow/nodes/decomposition.py

DECOMPOSITION_PROMPT = """
Break down this complex question into simpler sub-questions that can be answered independently.

Original question: {query}

Instructions:
1. Identify the key information needs
2. Create 2-5 independent sub-questions
3. Each sub-question should be answerable with a single retrieval
4. Preserve the context from the original question
5. Return as JSON array

Example:
Original: "Compare Python and JavaScript for web development"
Sub-questions:
- "What are the key features of Python for web development?"
- "What are the key features of JavaScript for web development?"
- "What are the advantages of Python over JavaScript for web backends?"
- "What are the advantages of JavaScript over Python for web frontends?"

Return ONLY the JSON array of sub-questions.
"""

class QueryDecomposer:
    """Decomposes complex queries into sub-questions."""

    def __init__(self, llm_client, max_sub_questions: int = 5):
        self.llm = llm_client
        self.max_sub_questions = max_sub_questions

    async def decompose(self, query: str) -> list[str]:
        """Decompose query into sub-questions."""
        prompt = DECOMPOSITION_PROMPT.format(query=query)

        response = await self.llm.complete(
            messages=[{"role": "user", "content": prompt}],
            temperature=0.0,
            max_tokens=500,
        )

        try:
            import json
            sub_questions = json.loads(response.content)
            return sub_questions[:self.max_sub_questions]
        except json.JSONDecodeError:
            # Fallback: return original query
            return [query]

async def decomposition_node(state: RAGState) -> RAGState:
    """Decompose complex query into sub-questions."""
    strategy = state.get("strategy")

    if strategy not in (RoutingStrategy.MULTI_HOP, RoutingStrategy.COMPARISON, RoutingStrategy.AGGREGATION):
        # Not a multi-hop query, skip decomposition
        state["sub_questions"] = [state["query"]]
        return state

    decomposer = QueryDecomposer(state["llm_client"])
    sub_questions = await decomposer.decompose(state["query"])

    state["sub_questions"] = sub_questions
    state["original_query"] = state["query"]

    logger.info(
        "query_decomposed",
        original_query=state["query"],
        sub_question_count=len(sub_questions),
    )

    return state
```
