## Technical Design

```python
# tests/e2e/test_rag_pipeline.py

import pytest
import asyncio
import httpx

ORCHESTRATOR_URL = "http://localhost:8003"
INGESTION_URL = "http://localhost:8001"

# Canonical test documents with known answers
TEST_DOCUMENTS = [
    {
        "id": "test-doc-1",
        "title": "Python Basics",
        "content": "Python was created by Guido van Rossum and released in 1991. "
                   "It uses indentation for code blocks instead of braces.",
        "expected_queries": [
            {"query": "Who created Python?", "expected_substring": "Guido van Rossum"},
            {"query": "When was Python released?", "expected_substring": "1991"},
        ],
    },
    # ... more test documents
]

@pytest.fixture(scope="session")
async def setup_test_data():
    """Ingest test documents before running tests."""
    async with httpx.AsyncClient() as client:
        for doc in TEST_DOCUMENTS:
            response = await client.post(
                f"{INGESTION_URL}/api/v1/ingest",
                json={
                    "source_type": "text",
                    "content": doc["content"],
                    "title": doc["title"],
                    "tenant_id": "e2e-test-tenant",
                },
            )
            assert response.status_code == 202

        # Wait for ingestion to complete
        await asyncio.sleep(10)

    yield

    # Cleanup: delete test tenant data

@pytest.mark.asyncio
async def test_query_returns_expected_answer(setup_test_data):
    """Query should return answer containing expected information."""
    async with httpx.AsyncClient() as client:
        for doc in TEST_DOCUMENTS:
            for test_case in doc["expected_queries"]:
                response = await client.post(
                    f"{ORCHESTRATOR_URL}/api/v1/query",
                    json={
                        "query": test_case["query"],
                        "tenant_id": "e2e-test-tenant",
                    },
                    timeout=30.0,
                )

                assert response.status_code == 200
                data = response.json()
                assert test_case["expected_substring"] in data["response"]

@pytest.mark.asyncio
async def test_query_latency_within_bounds(setup_test_data):
    """Query latency should be within acceptable bounds."""
    async with httpx.AsyncClient() as client:
        import time
        start = time.time()

        response = await client.post(
            f"{ORCHESTRATOR_URL}/api/v1/query",
            json={"query": "What is Python?", "tenant_id": "e2e-test-tenant"},
        )

        latency = time.time() - start

        assert response.status_code == 200
        assert latency < 5.0  # 5 second max for E2E

@pytest.mark.asyncio
async def test_citations_include_source_documents(setup_test_data):
    """Response should include citations from ingested documents."""
    async with httpx.AsyncClient() as client:
        response = await client.post(
            f"{ORCHESTRATOR_URL}/api/v1/query",
            json={"query": "Who created Python?", "tenant_id": "e2e-test-tenant"},
        )

        data = response.json()
        assert len(data.get("citations", [])) > 0
```

### Docker Compose for E2E

```yaml
# tests/e2e/docker-compose.e2e.yaml
version: "3.8"

services:
  e2e-tests:
    build:
      context: ../..
      dockerfile: tests/e2e/Dockerfile
    depends_on:
      orchestrator:
        condition: service_healthy
      retrieval:
        condition: service_healthy
      ingestion:
        condition: service_healthy
    environment:
      - ORCHESTRATOR_URL=http://orchestrator:8003
      - INGESTION_URL=http://ingestion:8001
    command: pytest tests/e2e/ -v --tb=short

  # Include all service dependencies
  postgres:
    image: postgres:16
    # ...
```

### CI Configuration

```yaml
# .github/workflows/e2e-tests.yaml
name: E2E Tests

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  e2e:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Start services
        run: docker-compose -f tests/e2e/docker-compose.e2e.yaml up -d

      - name: Wait for services
        run: |
          for i in {1..30}; do
            if curl -s http://localhost:8003/health | grep -q healthy; then
              break
            fi
            sleep 5
          done

      - name: Run E2E tests
        run: docker-compose -f tests/e2e/docker-compose.e2e.yaml run e2e-tests

      - name: Collect logs on failure
        if: failure()
        run: docker-compose -f tests/e2e/docker-compose.e2e.yaml logs
```
