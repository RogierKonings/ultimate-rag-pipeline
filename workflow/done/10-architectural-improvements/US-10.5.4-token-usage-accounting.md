> **Initiative:** 5 - Cost-Aware Retrieval & Model Tiering
> **Priority:** P1
> **Effort:** Medium (2-3 days)

## Summary

Track per-tenant token usage for LLM and embedding operations to enable quotas and billing.

## User Story

**As a** platform operator
**I want** per-tenant token usage tracking
**So that** I can implement quotas and billing

## Acceptance Criteria

- [ ] Counter: `llm_tokens_total{tenant_id, model, type}` (prompt/completion)
- [ ] Counter: `embeddings_generated_total{tenant_id}`
- [ ] Daily/monthly aggregation in PostgreSQL
- [ ] API endpoint for usage stats per tenant
- [ ] Optional: enforce quota limits with 429 response

```python
# services/shared/database/models/usage.py

class TokenUsage(Base, TimestampMixin):
    """Daily token usage aggregation."""
    __tablename__ = "token_usage"

    id: Mapped[str] = mapped_column(String(36), primary_key=True)
    tenant_id: Mapped[str] = mapped_column(String(100), index=True)
    date: Mapped[date] = mapped_column(Date, index=True)
    model: Mapped[str] = mapped_column(String(100))
    prompt_tokens: Mapped[int] = mapped_column(BigInteger, default=0)
    completion_tokens: Mapped[int] = mapped_column(BigInteger, default=0)
    embedding_tokens: Mapped[int] = mapped_column(BigInteger, default=0)

    __table_args__ = (
        UniqueConstraint('tenant_id', 'date', 'model', name='uq_usage_tenant_date_model'),
    )

# services/orchestrator/usage/tracker.py

class UsageTracker:
    """Tracks token usage per tenant."""

    def __init__(self, redis: Redis, session_factory):
        self.redis = redis
        self.session_factory = session_factory

    async def record_llm_usage(
        self,
        tenant_id: str,
        model: str,
        prompt_tokens: int,
        completion_tokens: int,
    ) -> None:
        """Record LLM token usage."""
        # Increment Redis counter (fast)
        key = f"usage:{tenant_id}:{date.today()}:{model}"
        await self.redis.hincrby(key, "prompt", prompt_tokens)
        await self.redis.hincrby(key, "completion", completion_tokens)
        await self.redis.expire(key, 86400 * 2)

        # Update Prometheus
        llm_tokens_total.labels(
            tenant_id=tenant_id,
            model=model,
            type="prompt",
        ).inc(prompt_tokens)
        llm_tokens_total.labels(
            tenant_id=tenant_id,
            model=model,
            type="completion",
        ).inc(completion_tokens)

    async def check_quota(self, tenant_id: str) -> tuple[bool, int]:
        """Check if tenant is within quota. Returns (allowed, remaining)."""
        config = await self._get_tenant_quota(tenant_id)
        if not config.quota_enabled:
            return True, -1

        current = await self._get_current_usage(tenant_id)
        remaining = config.monthly_limit - current

        return remaining > 0, remaining

    async def flush_to_database(self) -> None:
        """Flush Redis counters to PostgreSQL (run periodically)."""
        async with self.session_factory() as session:
            pattern = "usage:*"
            async for key in self.redis.scan_iter(pattern):
                data = await self.redis.hgetall(key)
                _, tenant_id, date_str, model = key.decode().split(":")

                usage = await session.execute(
                    select(TokenUsage).where(
                        TokenUsage.tenant_id == tenant_id,
                        TokenUsage.date == date.fromisoformat(date_str),
                        TokenUsage.model == model,
                    )
                )
                record = usage.scalar_one_or_none()

                if record:
                    record.prompt_tokens += int(data.get(b"prompt", 0))
                    record.completion_tokens += int(data.get(b"completion", 0))
                else:
                    session.add(TokenUsage(
                        id=str(uuid4()),
                        tenant_id=tenant_id,
                        date=date.fromisoformat(date_str),
                        model=model,
                        prompt_tokens=int(data.get(b"prompt", 0)),
                        completion_tokens=int(data.get(b"completion", 0)),
                    ))

            await session.commit()
```

### API Endpoint

```python
@router.get("/usage/{tenant_id}")
async def get_usage_stats(
    tenant_id: str,
    period: Literal["day", "week", "month"] = "month",
    session: AsyncSession = Depends(get_session),
) -> dict:
    """Get token usage statistics for tenant."""
    # Calculate date range
    end_date = date.today()
    if period == "day":
        start_date = end_date
    elif period == "week":
        start_date = end_date - timedelta(days=7)
    else:
        start_date = end_date - timedelta(days=30)

    result = await session.execute(
        select(
            TokenUsage.model,
            func.sum(TokenUsage.prompt_tokens).label("prompt_tokens"),
            func.sum(TokenUsage.completion_tokens).label("completion_tokens"),
        )
        .where(
            TokenUsage.tenant_id == tenant_id,
            TokenUsage.date >= start_date,
        )
        .group_by(TokenUsage.model)
    )

    return {
        "tenant_id": tenant_id,
        "period": period,
        "start_date": start_date.isoformat(),
        "end_date": end_date.isoformat(),
        "usage_by_model": [dict(row) for row in result],
    }
```
