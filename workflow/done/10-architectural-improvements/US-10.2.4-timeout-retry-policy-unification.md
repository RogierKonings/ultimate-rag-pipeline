# US-10.2.4: Timeout & Retry Policy Unification

> **Initiative:** 2 - End-to-End Resilience & Adaptive Degradation
> **Priority:** P0
> **Effort:** Small (1-2 days)
> **Status:** Draft

## Summary

Standardize timeout and retry policies across all services to ensure predictable, debuggable behavior and prevent cascading failures.

## Problem Statement

**Current state**: Timeout and retry configurations are scattered across services with inconsistent naming and values:
- Different env var patterns (`*_TIMEOUT`, `*_TIMEOUT_SECONDS`, `*_TIMEOUT_MS`)
- Unclear cascade behavior (what happens when inner timeout triggers)
- Some operations retry silently, others don't
- No central documentation of expected latencies

**Impact**:
- Difficult to debug timeout-related issues
- Cascading timeouts cause confusing errors
- Unpredictable behavior across environments
- Hard to tune for performance

## User Story

**As a** system developer
**I want** consistent timeout and retry policies across services
**So that** behavior is predictable and debuggable

### Retry Utility

```python
# services/shared/resilience/retry.py

import asyncio
import random
from functools import wraps
from typing import TypeVar, Callable, Any
import structlog

from shared.config.timeouts import TimeoutConfig

logger = structlog.get_logger(__name__)

T = TypeVar("T")

class RetryExhausted(Exception):
    """All retries exhausted."""
    def __init__(self, operation: str, attempts: int, last_error: Exception):
        self.operation = operation
        self.attempts = attempts
        self.last_error = last_error
        super().__init__(f"{operation} failed after {attempts} attempts: {last_error}")

async def with_retry(
    func: Callable[..., T],
    config: TimeoutConfig,
    operation_name: str,
    *args,
    **kwargs,
) -> T:
    """
    Execute function with timeout and retry.

    Args:
        func: Async function to call
        config: Timeout configuration
        operation_name: Name for logging
        *args, **kwargs: Arguments to pass to func

    Returns:
        Result from func

    Raises:
        RetryExhausted: If all retries fail
        asyncio.TimeoutError: If timeout exceeded
    """
    last_error = None
    attempts = config.retries + 1  # Initial attempt + retries

    for attempt in range(attempts):
        try:
            # Apply timeout
            result = await asyncio.wait_for(
                func(*args, **kwargs),
                timeout=config.timeout_ms / 1000.0,
            )

            if attempt > 0:
                logger.info(
                    "retry_succeeded",
                    operation=operation_name,
                    attempt=attempt + 1,
                )

            return result

        except asyncio.TimeoutError as e:
            last_error = e
            logger.warning(
                "operation_timeout",
                operation=operation_name,
                attempt=attempt + 1,
                timeout_ms=config.timeout_ms,
            )

        except Exception as e:
            last_error = e
            logger.warning(
                "operation_failed",
                operation=operation_name,
                attempt=attempt + 1,
                error=str(e),
            )

        # Calculate backoff if retrying
        if attempt < attempts - 1:
            backoff = _calculate_backoff(
                attempt=attempt,
                base_ms=config.backoff_base_ms,
                max_ms=config.backoff_max_ms,
            )
            logger.debug(
                "retry_backoff",
                operation=operation_name,
                backoff_ms=backoff,
            )
            await asyncio.sleep(backoff / 1000.0)

    raise RetryExhausted(operation_name, attempts, last_error)

def _calculate_backoff(attempt: int, base_ms: int, max_ms: int) -> int:
    """Calculate exponential backoff with jitter."""
    # Exponential backoff: base * 2^attempt
    backoff = base_ms * (2 ** attempt)
    # Cap at max
    backoff = min(backoff, max_ms)
    # Add jitter (±25%)
    jitter = backoff * 0.25 * (random.random() * 2 - 1)
    return int(backoff + jitter)

def retry_on_timeout(config: TimeoutConfig, operation_name: str):
    """
    Decorator for retry with timeout.

    Usage:
        @retry_on_timeout(RETRIEVAL_QDRANT_TIMEOUT, "qdrant_search")
        async def search_qdrant(query):
            ...
    """
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @wraps(func)
        async def wrapper(*args, **kwargs) -> T:
            return await with_retry(func, config, operation_name, *args, **kwargs)
        return wrapper
    return decorator
```

### Validation at Startup

```python
# services/shared/config/validate.py

from shared.config.timeouts import ALL_TIMEOUTS
import structlog

logger = structlog.get_logger(__name__)

def validate_timeout_cascade() -> list[str]:
    """
    Validate that timeout cascade is properly configured.

    Returns list of validation errors (empty if valid).
    """
    errors = []

    # Retrieval cascade
    retrieval_inner = max(
        ALL_TIMEOUTS["retrieval_embedding"].timeout_ms,
        ALL_TIMEOUTS["retrieval_qdrant"].timeout_ms,
        ALL_TIMEOUTS["retrieval_opensearch"].timeout_ms,
        ALL_TIMEOUTS["retrieval_reranker"].timeout_ms,
    )
    retrieval_outer = ALL_TIMEOUTS["retrieval_total"].timeout_ms

    if retrieval_inner >= retrieval_outer:
        errors.append(
            f"Retrieval inner timeout ({retrieval_inner}ms) >= "
            f"outer timeout ({retrieval_outer}ms)"
        )

    # Orchestrator cascade
    orch_retrieval = ALL_TIMEOUTS["orchestrator_retrieval"].timeout_ms
    orch_total = ALL_TIMEOUTS["orchestrator_total"].timeout_ms

    if orch_retrieval >= orch_total:
        errors.append(
            f"Orchestrator retrieval timeout ({orch_retrieval}ms) >= "
            f"total timeout ({orch_total}ms)"
        )

    # Orchestrator should allow time for retrieval
    retrieval_total = ALL_TIMEOUTS["retrieval_total"].timeout_ms
    if retrieval_total >= orch_retrieval:
        errors.append(
            f"Retrieval total ({retrieval_total}ms) >= "
            f"orchestrator retrieval timeout ({orch_retrieval}ms)"
        )

    if errors:
        for error in errors:
            logger.error("timeout_cascade_invalid", error=error)
    else:
        logger.info("timeout_cascade_valid")

    return errors

def validate_on_startup() -> None:
    """Run all configuration validations at startup."""
    errors = validate_timeout_cascade()
    if errors:
        raise ConfigurationError(f"Invalid timeout configuration: {errors}")
```

async def test_retry_succeeds_after_failures():
    """Should succeed if retry eventually works."""
    call_count = 0

    async def flaky_func():
        nonlocal call_count
        call_count += 1
        if call_count < 3:
            raise ConnectionError("Failed")
        return "success"

    config = TimeoutConfig(timeout_ms=1000, retries=3)
    result = await with_retry(flaky_func, config, "test_op")

    assert result == "success"
    assert call_count == 3

async def test_retry_exhausted():
    """Should raise RetryExhausted when all retries fail."""
    async def always_fails():
        raise ConnectionError("Always fails")

    config = TimeoutConfig(timeout_ms=1000, retries=2)

    with pytest.raises(RetryExhausted) as exc:
        await with_retry(always_fails, config, "test_op")

    assert exc.value.attempts == 3
    assert "Always fails" in str(exc.value.last_error)

async def test_timeout_triggers_retry():
    """Timeout should trigger retry."""
    call_count = 0

    async def slow_then_fast():
        nonlocal call_count
        call_count += 1
        if call_count == 1:
            await asyncio.sleep(10)  # Longer than timeout
        return "success"

    config = TimeoutConfig(timeout_ms=100, retries=1)
    result = await with_retry(slow_then_fast, config, "test_op")

    assert result == "success"
    assert call_count == 2
