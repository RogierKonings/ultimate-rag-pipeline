## Summary

Create a single source of truth for shared configuration values to prevent diverging defaults across services.

## User Story

**I want** a single source of truth for chunking and embedding config
**So that** services don't have diverging defaults

## Technical Design

```python
# services/shared/config/defaults.py

"""
Shared configuration defaults for RAG pipeline.

All services should import from this module to ensure consistency.
Environment variables override these defaults.
"""

from pydantic import BaseSettings, Field

class ChunkingConfig(BaseSettings):
    """Chunking configuration."""
    target_tokens: int = Field(300, env="CHUNKING_TARGET_TOKENS")
    max_tokens: int = Field(512, env="CHUNKING_MAX_TOKENS")
    overlap_tokens: int = Field(50, env="CHUNKING_OVERLAP_TOKENS")
    separators: list[str] = ["\n\n", "\n", ". ", " "]

    class Config:
        env_prefix = "CHUNKING_"

class EmbeddingConfig(BaseSettings):
    """Embedding configuration."""
    model_name: str = Field("BAAI/bge-large-en-v1.5", env="EMBEDDING_MODEL")
    dimensions: int = Field(1024, env="EMBEDDING_DIMENSIONS")
    batch_size: int = Field(32, env="EMBEDDING_BATCH_SIZE")
    normalize: bool = Field(True, env="EMBEDDING_NORMALIZE")
    query_prefix: str = Field(
        "Represent this sentence for searching relevant passages: ",
        env="EMBEDDING_QUERY_PREFIX"
    )

    class Config:
        env_prefix = "EMBEDDING_"

class RetrievalConfig(BaseSettings):
    """Retrieval configuration."""
    semantic_top_k: int = Field(50, env="RETRIEVAL_SEMANTIC_TOP_K")
    keyword_top_k: int = Field(50, env="RETRIEVAL_KEYWORD_TOP_K")
    rrf_k: int = Field(60, env="RETRIEVAL_RRF_K")
    semantic_weight: float = Field(0.7, env="RETRIEVAL_SEMANTIC_WEIGHT")
    keyword_weight: float = Field(0.3, env="RETRIEVAL_KEYWORD_WEIGHT")
    rerank_top_k: int = Field(10, env="RETRIEVAL_RERANK_TOP_K")
    reranker_model: str = Field("BAAI/bge-reranker-v2-m3", env="RERANKER_MODEL")

    class Config:
        env_prefix = "RETRIEVAL_"

# Singleton instances
chunking_config = ChunkingConfig()
embedding_config = EmbeddingConfig()
retrieval_config = RetrievalConfig()

def validate_config() -> list[str]:
    """Validate configuration at startup."""
    errors = []

    if embedding_config.dimensions not in [384, 768, 1024]:
        errors.append(f"Unusual embedding dimensions: {embedding_config.dimensions}")

    if chunking_config.overlap_tokens >= chunking_config.target_tokens:
        errors.append("Overlap tokens should be less than target tokens")

    if retrieval_config.semantic_weight + retrieval_config.keyword_weight != 1.0:
        errors.append("Semantic and keyword weights should sum to 1.0")

    return errors
```

### Service Usage

```python
# services/ingestion/chunking/splitter.py
from shared.config.defaults import chunking_config

class RecursiveTextSplitter:
    def __init__(self):
        self.target_tokens = chunking_config.target_tokens
        self.max_tokens = chunking_config.max_tokens
        self.overlap = chunking_config.overlap_tokens
        self.separators = chunking_config.separators


# services/retrieval/search/hybrid.py
from shared.config.defaults import retrieval_config

class HybridSearcher:
    def __init__(self):
        self.semantic_top_k = retrieval_config.semantic_top_k
        self.keyword_top_k = retrieval_config.keyword_top_k
        self.rrf_k = retrieval_config.rrf_k
```
