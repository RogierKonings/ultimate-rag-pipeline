# US-10.3.2: End-to-End Trace Hierarchy

> **Initiative:** 3 - Unified Cross-Service Observability
> **Priority:** P1
> **Effort:** Medium (2-3 days)
> **Status:** Draft

## Summary

Establish consistent span naming and parent-child relationships across all services, enabling complete visualization of request lifecycle in Jaeger/similar tools.

## Problem Statement

**Current state**: Services have OTEL instrumentation but:
- Span naming is inconsistent across services
- Parent-child relationships may not be properly established across HTTP boundaries
- No standard span hierarchy documentation
- Difficult to see complete request flow in trace UI

**Impact**:
- Trace visualizations are fragmented
- Hard to identify bottlenecks across services
- Inconsistent granularity of tracing
- Poor developer experience debugging

## User Story

**As a** performance engineer
**I want** to see a single trace from ingestion through retrieval to generation
**So that** I can identify bottlenecks across the pipeline

## Acceptance Criteria

# services/shared/observability/spans.py

"""
Span Naming Conventions for RAG Pipeline

All spans follow the pattern: {service}.{component}.{operation}

Services:
- ingestion
- retrieval
- orchestrator
- embedding

Components vary by service:
- ingestion: parse, chunk, embed, index
- retrieval: preprocess, semantic_search, keyword_search, fusion, rerank
- orchestrator: workflow, retrieval, generation

Examples:
- ingestion.parse.pdf
- retrieval.semantic_search.qdrant
- orchestrator.workflow.routing
"""

from enum import Enum

class SpanNames:
    """Canonical span names for consistent tracing."""

    # Ingestion Service
    INGESTION_DOCUMENT = "ingestion.document.process"
    INGESTION_PARSE = "ingestion.parse.{format}"  # pdf, docx, html, etc.
    INGESTION_CHUNK = "ingestion.chunk.split"
    INGESTION_EMBED = "ingestion.embed.batch"
    INGESTION_INDEX_QDRANT = "ingestion.index.qdrant"
    INGESTION_INDEX_OPENSEARCH = "ingestion.index.opensearch"
    INGESTION_INDEX_POSTGRES = "ingestion.index.postgres"

    # Retrieval Service
    RETRIEVAL_SEARCH = "retrieval.search.hybrid"
    RETRIEVAL_PREPROCESS = "retrieval.preprocess.query"
    RETRIEVAL_EMBED_QUERY = "retrieval.embed.query"
    RETRIEVAL_SEMANTIC = "retrieval.search.semantic"
    RETRIEVAL_KEYWORD = "retrieval.search.keyword"
    RETRIEVAL_FUSION = "retrieval.fusion.rrf"
    RETRIEVAL_RERANK = "retrieval.rerank.crossencoder"
    RETRIEVAL_ACL = "retrieval.filter.acl"

    # Orchestrator Service
    ORCHESTRATOR_QUERY = "orchestrator.query.process"
    ORCHESTRATOR_WORKFLOW = "orchestrator.workflow.{node}"

### OTEL Tracing Setup

```python
# services/shared/observability/tracing.py

from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter
from opentelemetry.instrumentation.httpx import HTTPXClientInstrumentation
from opentelemetry.instrumentation.asyncpg import AsyncPGInstrumentation
from opentelemetry.instrumentation.redis import RedisInstrumentation
from opentelemetry.propagate import set_global_textmap
from opentelemetry.propagators.composite import CompositePropagator
from opentelemetry.propagators.b3 import B3MultiFormat
from opentelemetry.trace.propagation.tracecontext import TraceContextTextMapPropagator
import os

def setup_tracing(service_name: str) -> trace.Tracer:
    """
    Configure OpenTelemetry tracing for a service.

    Sets up:
    - OTLP exporter to Jaeger/collector
    - W3C trace context propagation
    - Auto-instrumentation for httpx, asyncpg, redis
    """
    # Create tracer provider
    provider = TracerProvider()

    # Configure exporter
    otlp_endpoint = os.getenv("OTEL_EXPORTER_OTLP_ENDPOINT", "http://localhost:4317")
    exporter = OTLPSpanExporter(endpoint=otlp_endpoint)
    provider.add_span_processor(BatchSpanProcessor(exporter))

    # Set as global provider
    trace.set_tracer_provider(provider)

    # Configure propagation (W3C TraceContext + B3 for compatibility)
    set_global_textmap(
        CompositePropagator([
            TraceContextTextMapPropagator(),
            B3MultiFormat(),
        ])
    )

    # Auto-instrument libraries
    HTTPXClientInstrumentation().instrument()
    AsyncPGInstrumentation().instrument()
    RedisInstrumentation().instrument()

    return trace.get_tracer(service_name)
```

### Traced Qdrant Client

```python
# services/shared/vectorstore/traced_qdrant.py

from opentelemetry import trace
from opentelemetry.trace import SpanKind
from qdrant_client import AsyncQdrantClient
from shared.observability.spans import SpanNames
from functools import wraps

tracer = trace.get_tracer(__name__)

def trace_qdrant_operation(operation_name: str):
    """Decorator to add tracing to Qdrant operations."""
    def decorator(func):
        @wraps(func)
        async def wrapper(self, *args, **kwargs):
            with tracer.start_as_current_span(
                operation_name,
                kind=SpanKind.CLIENT,
                attributes={
                    "db.system": "qdrant",
                    "db.operation": func.__name__,
                    "db.collection": kwargs.get("collection_name", self.collection_name),
                },
            ) as span:
                try:
                    result = await func(self, *args, **kwargs)
                    if hasattr(result, "points"):
                        span.set_attribute("db.response_size", len(result.points))
                    return result
                except Exception as e:
                    span.set_attribute("error", True)
                    span.set_attribute("error.message", str(e))
                    raise
        return wrapper
    return decorator

class TracedQdrantClient:
    """Qdrant client with OpenTelemetry tracing."""

    def __init__(self, client: AsyncQdrantClient, collection_name: str):
        self._client = client
        self.collection_name = collection_name

    @trace_qdrant_operation(SpanNames.QDRANT_QUERY)
    async def query_points(
        self,
        collection_name: str | None = None,
        query: list[float],
        query_filter=None,
        limit: int = 10,
        **kwargs,
    ):
        """Search with tracing."""
        return await self._client.query_points(
            collection_name=collection_name or self.collection_name,
            query=query,
            query_filter=query_filter,
            limit=limit,
            **kwargs,
        )

    @trace_qdrant_operation(SpanNames.QDRANT_UPSERT)
    async def upsert(
        self,
        collection_name: str | None = None,
        points: list = None,
        **kwargs,
    ):
        """Upsert with tracing."""
        return await self._client.upsert(
            collection_name=collection_name or self.collection_name,
            points=points,
            **kwargs,
        )
```

# services/shared/search/traced_opensearch.py

from opentelemetry import trace
from opentelemetry.trace import SpanKind
from opensearchpy import AsyncOpenSearch
from shared.observability.spans import SpanNames

tracer = trace.get_tracer(__name__)

class TracedOpenSearchClient:
    """OpenSearch client with OpenTelemetry tracing."""

    def __init__(self, client: AsyncOpenSearch, index_name: str):
        self._client = client
        self.index_name = index_name

    async def search(
        self,
        index: str | None = None,
        body: dict = None,
        **kwargs,
    ):
        """Search with tracing."""
        with tracer.start_as_current_span(
            SpanNames.OPENSEARCH_QUERY,
            kind=SpanKind.CLIENT,
            attributes={
                "db.system": "opensearch",
                "db.operation": "search",
                "db.elasticsearch.index": index or self.index_name,
            },
        ) as span:
            try:
                result = await self._client.search(
                    index=index or self.index_name,
                    body=body,
                    **kwargs,
                )
                span.set_attribute(
                    "db.response_size",
                    result.get("hits", {}).get("total", {}).get("value", 0)
                )
                return result
            except Exception as e:
                span.set_attribute("error", True)
                span.set_attribute("error.message", str(e))
                raise

    async def index(
        self,
        index: str | None = None,
        body: dict = None,
        **kwargs,
    ):
        """Index with tracing."""
        with tracer.start_as_current_span(
            SpanNames.OPENSEARCH_INDEX,
            kind=SpanKind.CLIENT,
            attributes={
                "db.system": "opensearch",
                "db.operation": "index",
                "db.elasticsearch.index": index or self.index_name,
            },
        ):
            return await self._client.index(
                index=index or self.index_name,
                body=body,
                **kwargs,
            )

### Retrieval Service Tracing

```python
# services/retrieval/search/traced_hybrid.py

from opentelemetry import trace
from shared.observability.spans import SpanNames

tracer = trace.get_tracer("retrieval")

class TracedHybridSearcher:
    """Hybrid searcher with comprehensive tracing."""

    async def search(self, query: str, user_context, options) -> SearchResult:
        """Execute hybrid search with tracing."""
        with tracer.start_as_current_span(
            SpanNames.RETRIEVAL_SEARCH,
            attributes={
                "retrieval.query_length": len(query),
                "retrieval.tenant_id": user_context.tenant_id,
            },
        ) as root_span:

            # Preprocessing
            with tracer.start_as_current_span(SpanNames.RETRIEVAL_PREPROCESS):
                processed = await self.preprocessor.process(query)

            # Query embedding
            with tracer.start_as_current_span(
                SpanNames.RETRIEVAL_EMBED_QUERY,
                attributes={"embedding.model": self.embedding_model},
            ):
                embedding = await self.embedder.embed(processed.query)

            # Parallel search
            with tracer.start_as_current_span(SpanNames.RETRIEVAL_SEMANTIC) as sem_span:
                semantic_results = await self.semantic_searcher.search(
                    embedding, user_context, options.semantic_top_k
                )
                sem_span.set_attribute("retrieval.result_count", len(semantic_results))

            with tracer.start_as_current_span(SpanNames.RETRIEVAL_KEYWORD) as kw_span:
                keyword_results = await self.keyword_searcher.search(
                    processed.query, user_context, options.keyword_top_k
                )
                kw_span.set_attribute("retrieval.result_count", len(keyword_results))

            # Fusion
            with tracer.start_as_current_span(
                SpanNames.RETRIEVAL_FUSION,
                attributes={"fusion.method": "rrf", "fusion.k": 60},
            ):
                fused = self.fusion.fuse(semantic_results, keyword_results)

            # Reranking
            if options.use_reranker:
                with tracer.start_as_current_span(
                    SpanNames.RETRIEVAL_RERANK,
                    attributes={
                        "rerank.model": self.reranker_model,
                        "rerank.input_size": len(fused),
                    },
                ) as rerank_span:
                    final = await self.reranker.rerank(query, fused, options.rerank_top_k)
                    rerank_span.set_attribute("rerank.output_size", len(final))
            else:
                final = fused[:options.final_top_k]

            # ACL filtering
            with tracer.start_as_current_span(SpanNames.RETRIEVAL_ACL):
                filtered = self.acl_filter.filter(final, user_context)

            root_span.set_attribute("retrieval.final_count", len(filtered))
            return SearchResult(results=filtered, ...)
```

### Orchestrator Workflow Tracing

```python
# services/orchestrator/workflow/traced_workflow.py

from opentelemetry import trace
from shared.observability.spans import SpanNames

tracer = trace.get_tracer("orchestrator")

class TracedRAGWorkflow:
    """RAG workflow with comprehensive tracing."""

    async def execute(self, state: RAGState) -> RAGState:
        """Execute workflow with tracing."""
        with tracer.start_as_current_span(
            SpanNames.ORCHESTRATOR_QUERY,
            attributes={
                "orchestrator.query_length": len(state["query"]),
                "orchestrator.tenant_id": state.get("tenant_id"),
                "orchestrator.session_id": state.get("session_id"),
            },
        ) as root_span:

            # Routing
            with tracer.start_as_current_span(SpanNames.ORCHESTRATOR_ROUTING):
                state = await self.routing_node(state)
                root_span.set_attribute("orchestrator.strategy", state.get("strategy"))

            # Retrieval (if needed)
            if state.get("strategy") != "no_retrieval":
                with tracer.start_as_current_span(SpanNames.ORCHESTRATOR_RETRIEVAL):
                    state = await self.retrieval_node(state)

            # Prompt building
            with tracer.start_as_current_span(SpanNames.ORCHESTRATOR_PROMPT):
                state = await self.prompt_building_node(state)

            # Generation
            with tracer.start_as_current_span(
                SpanNames.ORCHESTRATOR_GENERATION,
                attributes={"generation.model": state.get("model_used")},
            ) as gen_span:
                state = await self.generation_node(state)
                gen_span.set_attribute("generation.tokens", state["usage"].get("total_tokens"))

            # Validation
            with tracer.start_as_current_span(SpanNames.ORCHESTRATOR_VALIDATION):
                state = await self.validation_node(state)

            return state
```
