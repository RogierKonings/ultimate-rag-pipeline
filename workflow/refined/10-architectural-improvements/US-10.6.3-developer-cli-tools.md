```python
#!/usr/bin/env python
# scripts/dev-query.py

"""CLI tool for testing RAG queries with debug output."""

import argparse
import asyncio
import httpx
import json
from rich.console import Console
from rich.table import Table

console = Console()

async def query(
    query_text: str,
    tenant_id: str = "dev-tenant",
    debug: bool = False,
    stream: bool = False,
):
    """Execute a RAG query with optional debug output."""
    url = "http://localhost:8003/api/v1/query"

    async with httpx.AsyncClient(timeout=60.0) as client:
        response = await client.post(
            url,
            json={
                "query": query_text,
                "tenant_id": tenant_id,
                "options": {"debug": debug},
            },
        )

        data = response.json()

        # Display response
        console.print("\n[bold green]Response:[/bold green]")
        console.print(data.get("response", "No response"))

        # Display citations
        if data.get("citations"):
            console.print("\n[bold blue]Citations:[/bold blue]")
            for i, cite in enumerate(data["citations"], 1):
                console.print(f"  [{i}] {cite.get('source', 'Unknown')}")

        # Display debug info
        if debug and data.get("debug"):
            console.print("\n[bold yellow]Debug Info:[/bold yellow]")
            debug_info = data["debug"]

            table = Table(title="Timing")
            table.add_column("Component")
            table.add_column("Latency (ms)")

            for component, latency in debug_info.get("timing", {}).items():
                table.add_row(component, f"{latency:.1f}")

            console.print(table)

            if debug_info.get("retrieval"):
                console.print(f"\nRetrieval: {debug_info['retrieval']}")

def main():
    parser = argparse.ArgumentParser(description="Test RAG queries")
    parser.add_argument("query", help="Query text")
    parser.add_argument("--tenant", default="dev-tenant", help="Tenant ID")
    parser.add_argument("--debug", action="store_true", help="Enable debug output")
    parser.add_argument("--stream", action="store_true", help="Stream response")

    args = parser.parse_args()
    asyncio.run(query(args.query, args.tenant, args.debug, args.stream))

if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python
# scripts/dev-health.py

"""Check health of all RAG pipeline services."""

import asyncio
import httpx
from rich.console import Console
from rich.table import Table

console = Console()

SERVICES = {
    "Orchestrator": "http://localhost:8003/health",
    "Retrieval": "http://localhost:8002/health",
    "Ingestion": "http://localhost:8001/health",
    "Embedding": "http://localhost:8080/health",
    "Qdrant": "http://localhost:6333/health",
    "OpenSearch": "http://localhost:9200/_cluster/health",
}

async def check_all():
    """Check health of all services."""
    table = Table(title="Service Health")
    table.add_column("Service")
    table.add_column("Status")
    table.add_column("Details")

    async with httpx.AsyncClient(timeout=5.0) as client:
        for name, url in SERVICES.items():
            try:
                response = await client.get(url)
                if response.status_code == 200:
                    data = response.json()
                    status = "[green]Healthy[/green]"
                    details = data.get("status", "OK")
                else:
                    status = "[yellow]Degraded[/yellow]"
                    details = f"HTTP {response.status_code}"
            except Exception as e:
                status = "[red]Unhealthy[/red]"
                details = str(e)[:50]

            table.add_row(name, status, details)

    console.print(table)

if __name__ == "__main__":
    asyncio.run(check_all())
```

```python
#!/usr/bin/env python
# scripts/dev-ingest.py

"""CLI tool for document ingestion."""

import argparse
import asyncio
import httpx
from pathlib import Path

async def ingest_file(
    file_path: str,
    tenant_id: str = "dev-tenant",
):
    """Ingest a file into the RAG pipeline."""
    path = Path(file_path)
    if not path.exists():
        print(f"File not found: {file_path}")
        return

    async with httpx.AsyncClient(timeout=60.0) as client:
        files = {"file": (path.name, path.read_bytes())}
        response = await client.post(
            "http://localhost:8001/api/v1/ingest/file",
            files=files,
            data={"tenant_id": tenant_id},
        )

        print(f"Status: {response.status_code}")
        print(f"Response: {response.json()}")

def main():
    parser = argparse.ArgumentParser(description="Ingest documents")
    parser.add_argument("file", help="File to ingest")
    parser.add_argument("--tenant", default="dev-tenant")

    args = parser.parse_args()
    asyncio.run(ingest_file(args.file, args.tenant))

if __name__ == "__main__":
    main()
```
