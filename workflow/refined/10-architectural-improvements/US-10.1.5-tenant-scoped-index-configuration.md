# US-10.1.5: Tenant-Scoped Index Configuration

> **Initiative:** 1 - Multi-Store Indexing & ACL Bulletproofing
> **Priority:** P2 (Medium-term)
> **Effort:** Large (5-7 days)
> **Status:** Draft

## Summary

Enable per-tenant collection/index isolation for large tenants, allowing independent scaling and workload isolation while maintaining shared infrastructure for smaller tenants.

## Problem Statement

**Current state**: All tenants share a single Qdrant collection (`documents`) and OpenSearch index, filtered by `tenant_id`:
- Simple to operate but creates contention at scale
- Large tenants affect query performance for others
- No ability to scale resources per tenant
- Backup/restore is all-or-nothing

**When this becomes necessary**:
- Tenant exceeds ~10M chunks
- Tenant requires dedicated SLAs
- Regulatory requirements for data isolation
- Tenant-specific tuning (different HNSW params)

## User Story

**As a** platform operator
**I want** the option to use per-tenant collections for large tenants
**So that** I can isolate workloads and scale independently

## Acceptance Criteria

```python
from enum import Enum
from sqlalchemy import String, JSON, Enum as SQLEnum
from sqlalchemy.orm import Mapped, mapped_column

class TenantIsolationMode(str, Enum):
    """Index isolation mode for tenant."""
    SHARED = "shared"      # Use shared collection/index (default)
    DEDICATED = "dedicated"  # Use tenant-specific collection/index

class TenantConfig(Base, TimestampMixin):
    """Tenant-level configuration."""
    __tablename__ = "tenant_configs"

    id: Mapped[str] = mapped_column(String(36), primary_key=True)
    tenant_id: Mapped[str] = mapped_column(
        String(100),
        unique=True,
        nullable=False,
        index=True,
    )

    # Index isolation
    isolation_mode: Mapped[TenantIsolationMode] = mapped_column(
        SQLEnum(TenantIsolationMode),
        default=TenantIsolationMode.SHARED,
        nullable=False,
    )

    # Qdrant settings (for dedicated collections)
    qdrant_collection_name: Mapped[str | None] = mapped_column(
        String(255),
        nullable=True,
        comment="Custom collection name (null = use default)",
    )
    qdrant_settings: Mapped[dict | None] = mapped_column(
        JSON,
        nullable=True,
        comment="Custom HNSW/optimization settings",
    )

    # OpenSearch settings
    opensearch_index_name: Mapped[str | None] = mapped_column(
        String(255),
        nullable=True,
    )
    opensearch_settings: Mapped[dict | None] = mapped_column(
        JSON,
        nullable=True,
    )

    # Tier configuration (for US-10.5.1)
    tier: Mapped[str] = mapped_column(
        String(20),
        default="standard",
        nullable=False,
    )

    def get_qdrant_collection(self) -> str:
        """Get Qdrant collection name for this tenant."""
        if self.isolation_mode == TenantIsolationMode.DEDICATED:
            return self.qdrant_collection_name or f"documents_{self.tenant_id}"
        return "documents"  # Shared collection

    def get_opensearch_index(self) -> str:
        """Get OpenSearch index name for this tenant."""
        if self.isolation_mode == TenantIsolationMode.DEDICATED:
            return self.opensearch_index_name or f"documents-{self.tenant_id}"
        return "documents"  # Shared index
```

### Collection Manager

```python
# services/shared/vectorstore/collection_manager.py

from qdrant_client import QdrantClient
from qdrant_client.models import (
    VectorParams, Distance, HnswConfigDiff,
    OptimizersConfigDiff, CollectionStatus,
)
import structlog

logger = structlog.get_logger(__name__)

DEFAULT_QDRANT_SETTINGS = {
    "vector_size": 1024,
    "distance": Distance.COSINE,
    "hnsw_config": {
        "m": 16,
        "ef_construct": 100,
    },
    "optimizers_config": {
        "memmap_threshold": 20000,
    },
}

class CollectionManager:
    """Manages Qdrant collections for multi-tenant isolation."""

    def __init__(self, client: QdrantClient):
        self.client = client

    async def ensure_collection_exists(
        self,
        collection_name: str,
        settings: dict | None = None,
    ) -> bool:
        """
        Ensure collection exists, creating if necessary.

        Args:
            collection_name: Name of collection
            settings: Optional custom settings (merged with defaults)

        Returns:
            True if collection was created, False if already existed
        """
        # Check if exists
        collections = await self.client.get_collections()
        if any(c.name == collection_name for c in collections.collections):
            logger.info("collection_exists", collection_name=collection_name)
            return False

        # Merge settings
        effective_settings = {**DEFAULT_QDRANT_SETTINGS, **(settings or {})}

        # Create collection
        await self.client.create_collection(
            collection_name=collection_name,
            vectors_config=VectorParams(
                size=effective_settings["vector_size"],
                distance=effective_settings["distance"],
            ),
            hnsw_config=HnswConfigDiff(**effective_settings["hnsw_config"]),
            optimizers_config=OptimizersConfigDiff(
                **effective_settings["optimizers_config"]
            ),
        )

        logger.info(
            "collection_created",
            collection_name=collection_name,
            settings=effective_settings,
        )
        return True

    async def create_tenant_collection(
        self,
        tenant_id: str,
        settings: dict | None = None,
    ) -> str:
        """
        Create dedicated collection for tenant.

        Args:
            tenant_id: Tenant identifier
            settings: Optional custom HNSW/optimizer settings

        Returns:
            Collection name created
        """
        collection_name = f"documents_{tenant_id}"
        await self.ensure_collection_exists(collection_name, settings)
        return collection_name

    async def delete_tenant_collection(
        self,
        tenant_id: str,
    ) -> bool:
        """Delete tenant's dedicated collection."""
        collection_name = f"documents_{tenant_id}"

        try:
            await self.client.delete_collection(collection_name)
            logger.info("collection_deleted", collection_name=collection_name)
            return True
        except Exception as e:
            logger.error(
                "collection_delete_failed",
                collection_name=collection_name,
                error=str(e),
            )
            return False

    async def get_collection_stats(
        self,
        collection_name: str,
    ) -> dict:
        """Get collection statistics."""
        info = await self.client.get_collection(collection_name)
        return {
            "name": collection_name,
            "vectors_count": info.vectors_count,
            "points_count": info.points_count,
            "status": info.status.value,
            "config": {
                "vector_size": info.config.params.vectors.size,
                "distance": info.config.params.vectors.distance.value,
            },
        }
```

```python
from opensearchpy import AsyncOpenSearch
import structlog

logger = structlog.get_logger(__name__)

DEFAULT_INDEX_SETTINGS = {
    "settings": {
        "number_of_shards": 1,
        "number_of_replicas": 1,
        "analysis": {
            "analyzer": {
                "default": {
                    "type": "standard",
                    "stopwords": "_english_",
                }
            }
        },
    },
    "mappings": {
        "properties": {
            "chunk_id": {"type": "keyword"},
            "document_id": {"type": "keyword"},
            "tenant_id": {"type": "keyword"},
            "status": {"type": "keyword"},
            "content": {"type": "text", "analyzer": "default"},
            "title": {"type": "text", "boost": 2},
            "visibility": {"type": "keyword"},
            "allowed_groups": {"type": "keyword"},
            "allowed_users": {"type": "keyword"},
            "owner_id": {"type": "keyword"},
            "source_uri": {"type": "keyword"},
            "created_at": {"type": "date"},
        }
    },
}

class OpenSearchIndexManager:
    """Manages OpenSearch indices for multi-tenant isolation."""

    def __init__(self, client: AsyncOpenSearch):
        self.client = client

    async def ensure_index_exists(
        self,
        index_name: str,
        settings: dict | None = None,
    ) -> bool:
        """
        Ensure index exists, creating if necessary.

        Returns True if created, False if already existed.
        """
        exists = await self.client.indices.exists(index=index_name)
        if exists:
            logger.info("index_exists", index_name=index_name)
            return False

        # Merge settings
        effective_settings = self._merge_settings(
            DEFAULT_INDEX_SETTINGS,
            settings or {},
        )

        await self.client.indices.create(
            index=index_name,
            body=effective_settings,
        )

        logger.info(
            "index_created",
            index_name=index_name,
        )
        return True

    async def create_tenant_index(
        self,
        tenant_id: str,
        settings: dict | None = None,
    ) -> str:
        """Create dedicated index for tenant."""
        index_name = f"documents-{tenant_id}"
        await self.ensure_index_exists(index_name, settings)
        return index_name

    async def delete_tenant_index(
        self,
        tenant_id: str,
    ) -> bool:
        """Delete tenant's dedicated index."""
        index_name = f"documents-{tenant_id}"
        try:
            await self.client.indices.delete(index=index_name)
            logger.info("index_deleted", index_name=index_name)
            return True
        except Exception as e:
            logger.error(
                "index_delete_failed",
                index_name=index_name,
                error=str(e),
            )
            return False

    def _merge_settings(self, base: dict, override: dict) -> dict:
        """Deep merge index settings."""
        result = base.copy()
        for key, value in override.items():
            if key in result and isinstance(result[key], dict):
                result[key] = self._merge_settings(result[key], value)
            else:
                result[key] = value
        return result
```

```python
# services/shared/vectorstore/qdrant_client.py

class TenantAwareQdrantClient:
    """Qdrant client with tenant-based collection routing."""

    def __init__(
        self,
        client: QdrantClient,
        config_repo: TenantConfigRepository,
    ):
        self.client = client
        self.config_repo = config_repo
        self._collection_cache: dict[str, str] = {}

    async def get_collection_for_tenant(self, tenant_id: str) -> str:
        """Get the appropriate collection name for tenant."""
        # Check cache first
        if tenant_id in self._collection_cache:
            return self._collection_cache[tenant_id]

        # Load tenant config
        config = await self.config_repo.get_by_tenant_id(tenant_id)
        if config is None:
            collection = "documents"  # Default shared
        else:
            collection = config.get_qdrant_collection()

        # Cache for future lookups
        self._collection_cache[tenant_id] = collection
        return collection

    def invalidate_cache(self, tenant_id: str) -> None:
        """Invalidate cached collection name (call after config change)."""
        self._collection_cache.pop(tenant_id, None)

    async def upsert_chunks(
        self,
        chunks: list[Chunk],
        embeddings: list[list[float]],
        tenant_id: str,
    ) -> None:
        """Upsert chunks to tenant's collection."""
        collection = await self.get_collection_for_tenant(tenant_id)

        points = [
            PointStruct(
                id=str(chunk.id),
                vector=embedding,
                payload=self._build_payload(chunk),
            )
            for chunk, embedding in zip(chunks, embeddings)
        ]

        await self.client.upsert(
            collection_name=collection,
            points=points,
        )

    async def search(
        self,
        query_embedding: list[float],
        tenant_id: str,
        filter: Filter,
        top_k: int = 50,
    ) -> list[ScoredPoint]:
        """Search tenant's collection."""
        collection = await self.get_collection_for_tenant(tenant_id)

        results = await self.client.query_points(
            collection_name=collection,
            query=query_embedding,
            query_filter=filter,
            limit=top_k,
            with_payload=True,
        )

        return results.points
```

### Migration Tool

```python
# services/ingestion/tools/migrate_tenant.py

import asyncio
from dataclasses import dataclass
import structlog

logger = structlog.get_logger(__name__)

@dataclass
class MigrationResult:
    tenant_id: str
    source_collection: str
    target_collection: str
    chunks_migrated: int
    duration_seconds: float
    success: bool
    error: str | None = None

async def migrate_tenant_to_dedicated(
    tenant_id: str,
    batch_size: int = 1000,
    dry_run: bool = True,
) -> MigrationResult:
    """
    Migrate tenant from shared collection to dedicated.

    This is a zero-downtime migration:
    1. Create dedicated collection
    2. Copy all tenant's vectors to new collection
    3. Update tenant config to use dedicated
    4. (Optional) Clean up vectors from shared collection

    Args:
        tenant_id: Tenant to migrate
        batch_size: Points to copy per batch
        dry_run: If True, simulate without making changes

    Returns:
        MigrationResult with details
    """
    start_time = datetime.utcnow()

    logger.info(
        "tenant_migration_started",
        tenant_id=tenant_id,
        dry_run=dry_run,
    )

    try:
        # Step 1: Create dedicated collection
        collection_manager = CollectionManager(qdrant_client)
        target_collection = f"documents_{tenant_id}"

        if not dry_run:
            await collection_manager.ensure_collection_exists(target_collection)

        # Step 2: Copy vectors in batches
        source_collection = "documents"
        total_copied = 0
        offset = None

        while True:
            # Scroll through tenant's vectors
            points, offset = await qdrant_client.scroll(
                collection_name=source_collection,
                scroll_filter=Filter(
                    must=[
                        FieldCondition(
                            key="tenant_id",
                            match=MatchValue(value=tenant_id),
                        )
                    ]
                ),
                limit=batch_size,
                offset=offset,
                with_payload=True,
                with_vectors=True,
            )

            if not points:
                break

            if not dry_run:
                # Upsert to target collection
                await qdrant_client.upsert(
                    collection_name=target_collection,
                    points=points,
                )

            total_copied += len(points)
            logger.info(
                "migration_batch_complete",
                tenant_id=tenant_id,
                batch_size=len(points),
                total_copied=total_copied,
            )

            if offset is None:
                break

        # Step 3: Update tenant config
        if not dry_run:
            async with async_session_factory() as session:
                config = await session.get(TenantConfig, tenant_id)
                if config is None:
                    config = TenantConfig(
                        id=str(uuid4()),
                        tenant_id=tenant_id,
                    )
                    session.add(config)

                config.isolation_mode = TenantIsolationMode.DEDICATED
                config.qdrant_collection_name = target_collection
                await session.commit()

        duration = (datetime.utcnow() - start_time).total_seconds()

        logger.info(
            "tenant_migration_complete",
            tenant_id=tenant_id,
            chunks_migrated=total_copied,
            duration_seconds=duration,
            dry_run=dry_run,
        )

        return MigrationResult(
            tenant_id=tenant_id,
            source_collection=source_collection,
            target_collection=target_collection,
            chunks_migrated=total_copied,
            duration_seconds=duration,
            success=True,
        )

    except Exception as e:
        logger.error(
            "tenant_migration_failed",
            tenant_id=tenant_id,
            error=str(e),
        )
        return MigrationResult(
            tenant_id=tenant_id,
            source_collection="documents",
            target_collection=f"documents_{tenant_id}",
            chunks_migrated=0,
            duration_seconds=(datetime.utcnow() - start_time).total_seconds(),
            success=False,
            error=str(e),
        )


async def rollback_tenant_to_shared(
    tenant_id: str,
) -> bool:
    """
    Rollback tenant from dedicated to shared collection.

    This updates config only - data remains in both collections.
    """
    async with async_session_factory() as session:
        config = await session.execute(
            select(TenantConfig).where(TenantConfig.tenant_id == tenant_id)
        )
        config = config.scalar_one_or_none()

        if config:
            config.isolation_mode = TenantIsolationMode.SHARED
            await session.commit()
            logger.info("tenant_rollback_complete", tenant_id=tenant_id)
            return True

    return False
```

```python
# services/ingestion/api/admin_routes.py

@router.post("/admin/tenants/{tenant_id}/migrate-to-dedicated")
async def migrate_tenant_to_dedicated_endpoint(
    tenant_id: str,
    dry_run: bool = True,
    background_tasks: BackgroundTasks,
    current_user: User = Depends(require_admin),
) -> dict:
    """
    Migrate tenant to dedicated collection/index.

    This is a long-running operation that runs in the background.
    """
    from ingestion.tools.migrate_tenant import migrate_tenant_to_dedicated

    if dry_run:
        # Synchronous dry run
        result = await migrate_tenant_to_dedicated(tenant_id, dry_run=True)
        return asdict(result)
    else:
        # Background migration
        background_tasks.add_task(
            migrate_tenant_to_dedicated,
            tenant_id,
            dry_run=False,
        )
        return {
            "status": "migration_started",
            "tenant_id": tenant_id,
            "message": "Migration running in background. Check logs for progress.",
        }

@router.get("/admin/tenants/{tenant_id}/isolation-status")
async def get_tenant_isolation_status(
    tenant_id: str,
    session: AsyncSession = Depends(get_session),
    current_user: User = Depends(require_admin),
) -> dict:
    """Get tenant's current isolation configuration."""
    config = await session.execute(
        select(TenantConfig).where(TenantConfig.tenant_id == tenant_id)

    config = config.scalar_one_or_none()

    if config is None:
        return {
            "tenant_id": tenant_id,
            "isolation_mode": "shared",
            "qdrant_collection": "documents",
            "opensearch_index": "documents",
        }

    return {
        "tenant_id": tenant_id,
        "isolation_mode": config.isolation_mode.value,
        "qdrant_collection": config.get_qdrant_collection(),
        "opensearch_index": config.get_opensearch_index(),

            "qdrant": config.qdrant_settings,

    }
```
