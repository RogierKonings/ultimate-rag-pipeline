# Global flag to track if auto-instrumentation has been set up
_auto_instrumented = False


def setup_auto_instrumentation() -> None:
    """
    Activate all OTEL auto-instrumentors.

    This function is idempotent - calling it multiple times has no effect
    after the first call.

    Instruments:
    - httpx: HTTP client calls
    - asyncpg: PostgreSQL async queries
    - redis: Redis operations
    """
    global _auto_instrumented

    if _auto_instrumented:
        logger.debug("Auto-instrumentation already configured, skipping")
        return

    try:
        from opentelemetry.instrumentation.httpx import HTTPXClientInstrumentation

        HTTPXClientInstrumentation().instrument()
        logger.info("HTTPXClientInstrumentation enabled")
    except ImportError:
        logger.warning("opentelemetry-instrumentation-httpx not installed, skipping")
    except Exception as e:
        logger.warning(f"Failed to instrument httpx: {e}")

    try:
        from opentelemetry.instrumentation.asyncpg import AsyncPGInstrumentation

        AsyncPGInstrumentation().instrument()
        logger.info("AsyncPGInstrumentation enabled")
    except ImportError:
        logger.warning("opentelemetry-instrumentation-asyncpg not installed, skipping")
    except Exception as e:
        logger.warning(f"Failed to instrument asyncpg: {e}")

    try:
        from opentelemetry.instrumentation.redis import RedisInstrumentation

        RedisInstrumentation().instrument()
        logger.info("RedisInstrumentation enabled")
    except ImportError:
        logger.warning("opentelemetry-instrumentation-redis not installed, skipping")
    except Exception as e:
        logger.warning(f"Failed to instrument redis: {e}")

    _auto_instrumented = True
    logger.info("Auto-instrumentation setup complete")
```

# services/shared/observability/clients/traced_qdrant.py
"""
Traced Qdrant client wrapper.

Adds OpenTelemetry tracing to Qdrant operations for distributed tracing.
"""

from __future__ import annotations

from typing import TYPE_CHECKING, Any

from opentelemetry import trace
from opentelemetry.trace import SpanKind, StatusCode

from shared.observability.otel.span_names import SpanNames

if TYPE_CHECKING:
    from qdrant_client import AsyncQdrantClient
    from qdrant_client.models import Filter

tracer = trace.get_tracer(__name__)


class TracedQdrantClient:
    """
    Qdrant client wrapper with OpenTelemetry tracing.

    This wrapper adds spans for all Qdrant operations, enabling
    visibility into vector search performance in distributed traces.

    Usage:
        raw_client = AsyncQdrantClient(url="localhost:6333")
        client = TracedQdrantClient(raw_client, collection_name="documents")
        results = await client.query_points(query=embedding, limit=10)
    """

    def __init__(self, client: "AsyncQdrantClient", collection_name: str):
        """
        Initialize traced client.

        Args:
            client: The underlying AsyncQdrantClient instance
            collection_name: Default collection name for operations
        """
        self._client = client
        self.collection_name = collection_name

    @property
    def client(self) -> "AsyncQdrantClient":
        """Return the underlying Qdrant client."""
        return self._client

    async def query_points(
        self,
        query: list[float],
        collection_name: str | None = None,
        query_filter: "Filter | None" = None,
        limit: int = 10,
        with_payload: bool = True,
        with_vectors: bool = False,
        score_threshold: float | None = None,
        **kwargs: Any,
    ) -> Any:
        """
        Search for similar vectors with tracing.

        Args:
            query: Query vector
            collection_name: Override default collection
            query_filter: Qdrant filter conditions
            limit: Maximum results to return
            with_payload: Include payload in results
            with_vectors: Include vectors in results
            score_threshold: Minimum similarity score
            **kwargs: Additional arguments passed to underlying client

        Returns:
            Qdrant query response with scored points
        """
        coll = collection_name or self.collection_name

        with tracer.start_as_current_span(
            SpanNames.QDRANT_QUERY,
            kind=SpanKind.CLIENT,
            attributes={
                "db.system": "qdrant",
                "db.operation": "query_points",
                "db.collection": coll,
                "db.qdrant.limit": limit,
                "db.qdrant.with_payload": with_payload,
                "db.qdrant.with_vectors": with_vectors,
            },
        ) as span:
            try:
                result = await self._client.query_points(
                    collection_name=coll,
                    query=query,
                    query_filter=query_filter,
                    limit=limit,
                    with_payload=with_payload,
                    with_vectors=with_vectors,
                    score_threshold=score_threshold,
                    **kwargs,
                )

                # Record result size
                if hasattr(result, "points"):
                    span.set_attribute("db.response_size", len(result.points))
                    if result.points:
                        span.set_attribute(
                            "db.qdrant.top_score", result.points[0].score
                        )

                return result

            except Exception as e:
                span.set_status(StatusCode.ERROR, str(e))
                span.record_exception(e)
                raise

    async def upsert(
        self,
        points: list[Any],
        collection_name: str | None = None,
        wait: bool = True,
        **kwargs: Any,
    ) -> Any:
        """
        Upsert vectors with tracing.

        Args:
            points: Points to upsert
            collection_name: Override default collection
            wait: Wait for operation to complete
            **kwargs: Additional arguments passed to underlying client

        Returns:
            Qdrant upsert response
        """
        coll = collection_name or self.collection_name
        points_count = len(points) if points else 0

        with tracer.start_as_current_span(
            SpanNames.QDRANT_UPSERT,
            kind=SpanKind.CLIENT,
            attributes={
                "db.system": "qdrant",
                "db.operation": "upsert",
                "db.collection": coll,
                "db.qdrant.points_count": points_count,
                "db.qdrant.wait": wait,
            },
        ) as span:
            try:
                result = await self._client.upsert(
                    collection_name=coll,
                    points=points,
                    wait=wait,
                    **kwargs,
                )
                return result

            except Exception as e:
                span.set_status(StatusCode.ERROR, str(e))
                span.record_exception(e)
                raise

    async def delete(
        self,
        points_selector: Any,
        collection_name: str | None = None,
        wait: bool = True,
        **kwargs: Any,
    ) -> Any:
        """
        Delete vectors with tracing.

        Args:
            points_selector: Points to delete (IDs or filter)
            collection_name: Override default collection
            wait: Wait for operation to complete
            **kwargs: Additional arguments passed to underlying client

        Returns:
            Qdrant delete response
        """
        coll = collection_name or self.collection_name

        with tracer.start_as_current_span(
            SpanNames.QDRANT_DELETE,
            kind=SpanKind.CLIENT,
            attributes={
                "db.system": "qdrant",
                "db.operation": "delete",
                "db.collection": coll,
                "db.qdrant.wait": wait,
            },
        ) as span:
            try:
                result = await self._client.delete(
                    collection_name=coll,
                    points_selector=points_selector,
                    wait=wait,
                    **kwargs,
                )
                return result

            except Exception as e:
                span.set_status(StatusCode.ERROR, str(e))
                span.record_exception(e)
                raise
```

"""
Traced OpenSearch client wrapper.

Adds OpenTelemetry tracing to OpenSearch operations for distributed tracing.
"""

from __future__ import annotations

from typing import TYPE_CHECKING, Any

from opentelemetry import trace
from opentelemetry.trace import SpanKind, StatusCode

from shared.observability.otel.span_names import SpanNames

if TYPE_CHECKING:
    from opensearchpy import AsyncOpenSearch

tracer = trace.get_tracer(__name__)


class TracedOpenSearchClient:
    """
    OpenSearch client wrapper with OpenTelemetry tracing.

    This wrapper adds spans for all OpenSearch operations, enabling
    visibility into keyword search performance in distributed traces.

    Usage:
        raw_client = AsyncOpenSearch(hosts=["localhost:9200"])
        client = TracedOpenSearchClient(raw_client, index_name="documents")
        results = await client.search(body={"query": {"match": {"content": "test"}}})
    """

    def __init__(self, client: "AsyncOpenSearch", index_name: str):
        """
        Initialize traced client.

        Args:
            client: The underlying AsyncOpenSearch instance
            index_name: Default index name for operations
        """
        self._client = client
        self.index_name = index_name

    @property
    def client(self) -> "AsyncOpenSearch":
        """Return the underlying OpenSearch client."""
        return self._client

    async def search(
        self,
        body: dict[str, Any],
        index: str | None = None,
        **kwargs: Any,
    ) -> dict[str, Any]:
        """
        Search with tracing.

        Args:
            body: OpenSearch query DSL body
            index: Override default index
            **kwargs: Additional arguments passed to underlying client

        Returns:
            OpenSearch search response
        """
        idx = index or self.index_name

        with tracer.start_as_current_span(
            SpanNames.OPENSEARCH_QUERY,
            kind=SpanKind.CLIENT,
            attributes={
                "db.system": "opensearch",
                "db.operation": "search",
                "db.elasticsearch.index": idx,
            },
        ) as span:
            try:
                result = await self._client.search(index=idx, body=body, **kwargs)

                # Record result size
                hits = result.get("hits", {})
                total = hits.get("total", {})
                count = total.get("value", 0) if isinstance(total, dict) else total
                span.set_attribute("db.response_size", count)

                # Record top score if available
                if hits.get("hits"):
                    top_hit = hits["hits"][0]
                    if "_score" in top_hit and top_hit["_score"] is not None:
                        span.set_attribute("db.opensearch.top_score", top_hit["_score"])

                return result

            except Exception as e:
                span.set_status(StatusCode.ERROR, str(e))
                span.record_exception(e)
                raise

    async def index(
        self,
        body: dict[str, Any],
        index: str | None = None,
        id: str | None = None,
        **kwargs: Any,
    ) -> dict[str, Any]:
        """
        Index document with tracing.

        Args:
            body: Document to index
            index: Override default index
            id: Document ID
            **kwargs: Additional arguments passed to underlying client

        Returns:
            OpenSearch index response
        """
        idx = index or self.index_name

        with tracer.start_as_current_span(
            SpanNames.OPENSEARCH_INDEX,
            kind=SpanKind.CLIENT,
            attributes={
                "db.system": "opensearch",
                "db.operation": "index",
                "db.elasticsearch.index": idx,
            },
        ) as span:
            try:
                result = await self._client.index(
                    index=idx, body=body, id=id, **kwargs
                )
                return result

            except Exception as e:
                span.set_status(StatusCode.ERROR, str(e))
                span.record_exception(e)
                raise

    async def bulk(
        self,
        body: list[dict[str, Any]],
        index: str | None = None,
        **kwargs: Any,
    ) -> dict[str, Any]:
        """
        Bulk indexing with tracing.

        Args:
            body: Bulk operations body
            index: Default index for operations
            **kwargs: Additional arguments passed to underlying client

        Returns:
            OpenSearch bulk response
        """
        idx = index or self.index_name
        # Estimate operation count (bulk body alternates action/document)
        operation_count = len(body) // 2 if body else 0

        with tracer.start_as_current_span(
            SpanNames.OPENSEARCH_BULK,
            kind=SpanKind.CLIENT,
            attributes={
                "db.system": "opensearch",
                "db.operation": "bulk",
                "db.elasticsearch.index": idx,
                "db.opensearch.bulk_operations": operation_count,
            },
        ) as span:
            try:
                result = await self._client.bulk(body=body, index=idx, **kwargs)

                # Record if there were errors
                span.set_attribute("db.opensearch.bulk_errors", result.get("errors", False))

                return result

            except Exception as e:
                span.set_status(StatusCode.ERROR, str(e))
                span.record_exception(e)
                raise

    async def delete(
        self,
        id: str,
        index: str | None = None,
        **kwargs: Any,
    ) -> dict[str, Any]:
        """
        Delete document with tracing.

        Args:
            id: Document ID to delete
            index: Override default index
            **kwargs: Additional arguments passed to underlying client

        Returns:
            OpenSearch delete response
        """
        idx = index or self.index_name

        with tracer.start_as_current_span(
            SpanNames.OPENSEARCH_DELETE,
            kind=SpanKind.CLIENT,
            attributes={
                "db.system": "opensearch",
                "db.operation": "delete",
                "db.elasticsearch.index": idx,
            },
        ) as span:
            try:
                result = await self._client.delete(index=idx, id=id, **kwargs)
                return result

            except Exception as e:
                span.set_status(StatusCode.ERROR, str(e))
                span.record_exception(e)
                raise

async def search(
    self,
    query: str,
    query_embedding: list[float],
    top_k: int | None = None,
    filters: dict | None = None,
    config: HybridSearchConfig | None = None,
) -> HybridSearchResponse:
    """
    Execute hybrid search combining semantic and keyword search.
    """
    with tracer.start_as_current_span(
        SpanNames.RETRIEVAL_SEARCH,
        attributes={
            "retrieval.query_length": len(query),
            "retrieval.embedding_dim": len(query_embedding),
        },
    ) as root_span:
        start_time = time.time()

        # Use provided config or default
        cfg = config or self.config
        final_top_k = top_k or cfg.top_k

        # Create fusion for this config
        fusion = self._create_fusion(cfg) if config else self._fusion

        # Run both searches in parallel with child spans
        with tracer.start_as_current_span(SpanNames.RETRIEVAL_SEMANTIC):
            semantic_task = self.semantic.search(
                query_embedding=query_embedding,
                top_k=cfg.semantic_top_k,
                filters=filters,
                score_threshold=0.0,
            )

        with tracer.start_as_current_span(SpanNames.RETRIEVAL_KEYWORD):
            keyword_task = self.keyword.search(
                query=query,
                top_k=cfg.keyword_top_k,
                filters=filters,
                min_score=0.0,
            )

        semantic_response, keyword_response = await asyncio.gather(
            semantic_task,
            keyword_task,
        )

        # Fuse results
        with tracer.start_as_current_span(
            SpanNames.RETRIEVAL_FUSION,
            attributes={
                "fusion.semantic_count": len(semantic_response.results),
                "fusion.keyword_count": len(keyword_response.results),
            },
        ):
            fused_results = fusion.fuse(
                semantic_results=semantic_response.results,
                keyword_results=keyword_response.results,
                top_k=final_top_k,
            )

        # Apply score threshold
        if cfg.min_score > 0:
            fused_results = [r for r in fused_results if r.fused_score >= cfg.min_score]

        # Deduplicate if needed
        if cfg.deduplicate:
            fused_results = self._deduplicate(fused_results)

        search_time = (time.time() - start_time) * 1000

        root_span.set_attribute("retrieval.result_count", len(fused_results))
        root_span.set_attribute("retrieval.search_time_ms", search_time)

        return HybridSearchResponse(
            results=fused_results,
            total_semantic=semantic_response.total_found,
            total_keyword=keyword_response.total_found,
            search_time_ms=search_time,
            fusion_method=cfg.fusion_method,
        )
